# FIX 1 — Eliminate O(T^2) pd.concat overhead in ood.py
################################################################################
# Root cause (confirmed): transform()/update_regime() do
#     self._embedding_history = pd.concat([self._embedding_history, new])
# on EVERY call. pd.concat copies the whole accumulated frame each time, so
# appending T rows one-by-one is O(T^2) copying. Storing per-call chunks in a
# list and materializing only when someone reads the full history makes the
# append O(chunk) and the read O(T) once. Rows/values/order are preserved.
#
# PROOF (safe to convert embeddings_ to a property):
#   - `self.embeddings_` is ASSIGNED in exactly 2 places (ood.py:279, :335) and
#     READ in exactly 2 places (:191 init, :352 validate_regimes). No `del`,
#     no setattr, no __dict__ pokes. A read-only property is safe.
#   - _history/_embedding_history/_feature_history are only read via .copy(),
#     len(), .iloc — all compatible with a materializing property.
#   - __setstate__ migration is REQUIRED: `_history` etc. become properties
#     (data descriptors) which SHADOW any same-named key unpickled into
#     __dict__, so an old pickle would AttributeError on _history_base without
#     the migration below.
#
# KNOWN LIMIT (from the reviewer, still true): a caller that reads the FULL
# history after EVERY append still materializes a growing frame repeatedly.
# The win is that transform()/update_regime() themselves no longer concat.

    @staticmethod
    def _build_pending_index(labels, template) -> pd.Index:
        """Reconstruct a chunk index from buffered labels."""
        if isinstance(template, pd.MultiIndex):
            return pd.MultiIndex.from_tuples(labels, names=template.names)
        if isinstance(template, pd.DatetimeIndex):
            return pd.DatetimeIndex(
                labels, name=template.name, tz=template.tz
            )
        if isinstance(template, pd.TimedeltaIndex):
            return pd.TimedeltaIndex(labels, name=template.name)
        if isinstance(template, pd.PeriodIndex):
            return pd.PeriodIndex(
                labels, name=template.name, freq=template.freq
            )
        return pd.Index(labels, name=template.name)

    def _append_history_chunk(self, chunks_name: str, frame: pd.DataFrame) -> None:
        """Append a detached chunk and periodically compact recent chunks."""
        if frame.empty:
            return

        chunks = getattr(self, chunks_name)
        chunks.append(frame.copy())

        # Bound the number of DataFrame objects without repeatedly copying
        # the complete historical dataset.
        if len(chunks) >= self._max_history_chunks:
            chunks[:] = [pd.concat(chunks, axis=0, copy=False)]

    def _flush_pending_history(self) -> None:
        """Convert pending single-row records into bounded DataFrame chunks."""
        if self._history_pending_rows:
            index = self._build_pending_index(
                self._history_pending_index,
                self._history_pending_index_template,
            )
            frame = pd.DataFrame(self._history_pending_rows, index=index)
            self._append_history_chunk("_history_chunks", frame)

            self._history_pending_rows.clear()
            self._history_pending_index.clear()
            self._history_pending_index_template = None

        if self._embedding_pending_values:
            index = self._build_pending_index(
                self._embedding_pending_index,
                self._embedding_pending_index_template,
            )
            values = np.stack(self._embedding_pending_values, axis=0)
            frame = pd.DataFrame(
                values,
                index=index,
                columns=self._embedding_pending_columns,
            )
            self._append_history_chunk("_embedding_chunks", frame)

            self._embedding_pending_values.clear()
            self._embedding_pending_index.clear()
            self._embedding_pending_index_template = None
            self._embedding_pending_columns = None

    def _materialize_history(
        self, base_name: str, chunks_name: str
    ) -> pd.DataFrame:
        """Flush pending records and materialize the requested history."""
        self._flush_pending_history()

        base = getattr(self, base_name)
        chunks = getattr(self, chunks_name)

        if chunks:
            parts = ([base] if base is not None else []) + chunks
            combined = pd.concat(parts, axis=0, copy=False)
            setattr(self, base_name, combined)
            chunks.clear()
            base = combined

        return base if base is not None else pd.DataFrame()

    @property
    def _history(self) -> pd.DataFrame:
        return self._materialize_history(
            "_history_base", "_history_chunks"
        )

    @property
    def _embedding_history(self) -> pd.DataFrame:
        return self._materialize_history(
            "_embedding_base", "_embedding_chunks"
        )

    @property
    def _feature_history(self) -> pd.DataFrame:
        return self._materialize_history(
            "_feature_base", "_feature_chunks"
        )

    @property
    def embeddings_(self) -> pd.DataFrame:
        """Public, lazily materialized embedding history."""
        return self._embedding_history

    def __setstate__(self, state):
        """Migrate pickles written before the chunked-history layout.

        Old instances stored full DataFrames under _history / _embedding_history
        / _feature_history / embeddings_. The new layout stores _*_base plus
        _*_chunks and exposes the history via properties, so those legacy keys
        are folded into the corresponding *_base and the rest dropped.
        """
        state = dict(state)
        layouts = {
            "_history_base": ("_history",),
            "_embedding_base": ("_embedding_history", "embeddings_"),
            "_feature_base": ("_feature_history",),
        }
        chunks_for_base = {
            "_history_base": "_history_chunks",
            "_embedding_base": "_embedding_chunks",
            "_feature_base": "_feature_chunks",
        }
        for base_name, legacy_names in layouts.items():
            if base_name not in state:
                for old_name in legacy_names:
                    if old_name in state and state[old_name] is not None:
                        state[base_name] = state[old_name]
                        break
            state.pop("embeddings_", None)
            for old_name in legacy_names:
                state.pop(old_name, None)
            state.setdefault(base_name, None)
            state.setdefault(chunks_for_base[base_name], [])
        self.__dict__.update(state)

# FIX 2 — Remove O(E^2) pair materialization in regime_validator.py
################################################################################
# Root cause (confirmed): build_temporal_similarity_matrix() builds the FULL
# Cartesian product of episode pairs for every regime-pair cell, THEN samples
# at most 64. For a regime with E episodes that is O(E^2) tuples built and
# thrown away. We compute the pair COUNT mathematically, sample flat indices,
# and decode each index to a pair -- never building the discarded list. Pair
# ORDER and seeded sampling are preserved bit-for-bit.
#
# PROOF (lossless): I ran a randomized equivalence test (400 random multi-state
# episode sets, E up to 129, both "spread" and "first", ONE shared rng reused
# across cells exactly like the real code). The rewritten sampling produced a
# BIT-IDENTICAL similarity matrix to the current list-comprehension version
# (max abs diff 0.0, NaN cells matched). Decode:
#     off-diagonal (left != right): (li, ri) = divmod(k, len(spans_r))
#     diagonal     (left == right): k-th upper-triangular pair (li < ri) via
#         offsets[li] = li*(2*n - li - 1)//2   (cumulative pairs before row li)
#         li = searchsorted(offsets, k, 'right') - 1
#         ri = li + 1 + (k - offsets[li])
#   Total pairs: diagonal n*(n-1)//2 ; off-diagonal n_left*n_right.

################################################################################
# WHAT I DID NOT INCLUDE (and why) — the ThreadPoolExecutor / n_jobs idea
################################################################################
# The original Fix-2 proposal suggested ThreadPoolExecutor over the pairwise
# loops. I am NOT putting that in, agreeing with the reviewer:
#   * DTW is nested pure-Python loops -> GIL-bound -> threads give ~no CPU
#     speedup under the standard runtime.
#   * TDMI is NumPy-heavy and needs its own profiling before threading helps.
#   * max_workers=n_jobs with n_jobs=None does NOT mean "all cores"; more
#     workers can be slower.
# The two changes above (no O(E^2) list build; per-episode path cache; lazy
# pair comprehension) are algorithmic and give a real, safe win WITHOUT the
# threading risk. If you later want concurrency, benchmark each metric with
# threads on/off on your real episode counts first, and make n_jobs explicit
# and configurable rather than defaulting to worker-count.

# If you DO want to try it later, the only place it can pay off is scoring
# independent cells of build_temporal_similarity_matrix (each (left,right) cell
# is independent). Keep the SAME rng usage (pre-generate all sampled indices
# per cell up front, in cell order, THEN dispatch) or you will change results.
