 506|                 if is_buy {
 507|                     self.bar_retail_buy += amount;
 508|                 } else {
 509|                     self.bar_retail_sell += amount;
 510|                 }
 511|             } else if is_buy {
 512|                 self.bar_inst_buy += amount;
 513|             } else {
 514|                 self.bar_inst_sell += amount;
 515|             }

 517|             // Tick sign (tick rule) vs the previous TRADE price. `resample('1s')

 535|             self.last_trade_price = Some(price);
