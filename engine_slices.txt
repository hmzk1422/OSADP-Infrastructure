 603|     /// Advance the composite (TIMEFRAME_BARRIER) bar with a price tick from
 604|     /// either a trade print or an orderbook mid. On the first event of a new
 605|     /// composite bar the macro/regime features are recomputed once and then

 627|     fn push_price(&mut self, ts: i64, price: f64, trade: Option<(f64, bool)>) -> Vec<EngineEvent> {
 628|         let micro_ms = self.cfg.micro_barrier_secs * 1000;
 629|         let micro_idx = ts / micro_ms;
 630|         let mut events = Vec::new();
 631| 
 632|         // 1. Micro (1 s) bar rollover ΓåÆ finalize the completed 1 s bar with the
 633|         //    full feature vector (micro grid + forward-filled macro) and
 634|         //    evaluate a fresh entry from it. The model runs on every 1 s row.
 635|         match self.micro_idx {
 636|             None => self.start_micro(micro_idx, price),
 637|             Some(cur) if micro_idx > cur => {
 638|                 // The research state space is dense at one-second resolution.
 639|                 // Close every missing interval at the last trade price so a
 640|                 // sparse feed cannot change model cadence or timeout timing.
 641|                 let mut idx = cur;
 642|                 while idx < micro_idx {
 643|                     let bar_close = self.micro_close;
 644|                     let bar_ms = idx * micro_ms;
 645|                     let feat = self.finish_micro(bar_close, bar_ms);
 646|                     events.push(EngineEvent::BarClosed(feat));
 647| 
 648|                     // Research `_sequential_triple_barrier` checks barrier
 649|                     // crossings against the 1-SECOND BAR CLOSE series only
 650|                     // (sliding_window_view of 1s closes) ΓÇö never on an
 651|                     // intra-second wick. Evaluate exits on the same grid.
 652|                     if let Some(rec) = self.check_exit(bar_ms, bar_close) {
 653|                         events.push(EngineEvent::Exited(rec));
 654|                     }
 655| 
 656|                     self.evaluate_entry(&mut events);
 657|                     let next_idx = idx + 1;
 658|                     self.start_micro(next_idx, bar_close);
 659|                     idx = next_idx;
 660|                 }
 661|             }
 662|             Some(cur) if micro_idx < cur => return events,
 663|             Some(_) => {}
 664|         }
 665| 
 666|         // 2. Macro (regime) rollover + OHLC ΓÇö driven by price events from
 667|         //    EITHER source (trade print or orderbook mid). The first event of
 668|         //    a new 300 s bar recomputes the composite features ONCE from the
 669|         //    just-closed bar and forward-fills them to subsequent 1 s bars.
 670|         self.push_macro_price(ts, price);
 671| 
 672|         // 3. Update the in-progress micro bar OHLC (macro handled above).
 673|         if self.micro_idx == Some(micro_idx) {
 674|             self.micro_high = self.micro_high.max(price);
 675|             self.micro_low = self.micro_low.min(price);
 676|             self.micro_close = price;
 677|         }
 678| 
 679|         if trade.is_some() {

 681|         }
 682| 
 683|         // Exits are evaluated on 1-second bar closes only (see the rollover
 684|         // loop above) to match research's triple-barrier-on-1s-close semantics.
 685|         events
 686|     }

 777|     fn evaluate_entry(&mut self, events: &mut Vec<EngineEvent>) {
 778|         if self.position.is_some() {
 779|             return;
 780|         }
 781|         // Research-style lockout: after an exit, re-entry is blocked until
 782|         // entry_time + actual_duration + 1 s ΓÇö which equals `exit_ms + 1000`
 783|         // in both barrier and timeout cases (research computes
 784|         // `next_available_time = actual_global_time + (bars_to_exit + 1)`).
 785|         // NOTE this deliberately ignores the magnitude of LOCKOUT_SECONDS and
 786|         // only uses it as an on/off flag ΓÇö exactly like the research loop.
 787|         if self.cfg.lockout_seconds > 0 && self.last_bar_close_ms < self.next_available_entry_ms {
 788|             return;
 789|         }
 790|         let Some(feat) = self.last_features else {
 791|             return;
 792|         };
 793| 
 794|         // Vol filter mirrors `_passes_vol_filter` (MIN/MAX_BARRIER_PCT).
 795|         let bp = self.barrier_pct();
 796|         if bp < self.cfg.min_barrier_pct || bp > self.cfg.max_barrier_pct {
 797|             return;
 798|         }
 799| 
 800|         let (in_zone, side) = self.resolve_zone_side(&feat);
 801|         if !in_zone {
 802|             return;
 803|         }
 804| 
 805|         // Research `find_debounced_signals` is a PURE threshold on the model's
 806|         // P(WIN) ΓÇö `np.where(y_proba >= threshold)`. There is NO hysteresis /
 807|         // debounce state machine in the research backtest (the hysteresis arg
 808|         // is deliberately ignored there); the lockout filter alone governs
 809|         // trade spacing. Fire on EVERY in-zone 1 s bar close at/above the gate.
 810|         if feat.win_prob + 1e-12 < self.cfg.confidence_threshold {
 811|             return; // LOW_CONFIDENCE
 812|         }
 813| 
 814|         // Research fills at `execution_price` = the LAST TRADE PRICE (the
 815|         // state-space grid's unshifted micro-bar price). The engine's `mid`
 816|         // mixes book mid with last-trade price; prefer the pure trade-print
 817|         // fill to match the trained entry basis.
 818|         let fill_price = self
 819|             .last_trade_price
 820|             .filter(|p| *p > 0.0)
 821|             .or_else(|| if self.mid > 0.0 { Some(self.mid) } else { None })
 822|             .unwrap_or(feat.close);
 823|         self.enter_position(feat.ts, side, fill_price, bp);
 824| 
 825|         let p = self.position.as_ref().expect("position just opened");
 826|         events.push(EngineEvent::Entered(EntryReceipt {
 827|             ts: p.entry_ms,
 828|             side: p.side,
 829|             entry_mid: p.entry_mid,
 830|             entry_price: p.entry_price,
 831|             qty: p.qty,
 832|             notional: p.notional,
 833|             barrier_pct: p.barrier_pct,
 834|             stop: p.stop,
 835|             target: p.target,
 836|             rsi_2: feat.rsi_2,
 837|             win_prob: feat.win_prob,
 838|         }));
 839|     }

 841|     /// Resolve the hunting zone + trade side exactly like research
 842|     /// `_infer_trade_side` (+ executor's zone rules).
 843|     fn resolve_zone_side(&self, feat: &Feature) -> (bool, Side) {
 844|         let thr = self.cfg.rsi_filter_threshold;
 845|         let high_thr = 100.0 - thr;
 846|         let dir_side = if self.cfg.trade_direction_is_long {
 847|             Side::Long
 848|         } else {
 849|             Side::Short
 850|         };
 851| 
 852|         if self.cfg.enable_rsi_accel_filter {
 853|             // Two-sided directional zone: (rsi<thr & acc>0)ΓåÆlong, (rsi>100-thr
 854|             // & acc<0)ΓåÆshort, then flipped by TRADE_DIRECTION.
 855|             let zone_side = if feat.rsi_2 < thr && feat.ma_accel > 0.0 {
 856|                 1
 857|             } else if feat.rsi_2 > high_thr && feat.ma_accel < 0.0 {
 858|                 -1
 859|             } else {
 860|                 return (false, dir_side);
 861|             };
 862|             let side = if (zone_side == 1) == self.cfg.trade_direction_is_long {
 863|                 Side::Long
 864|             } else {
 865|                 Side::Short
 866|             };
 867|             (true, side)
 868|         } else {
 869|             // Single zone: rsi_2 < threshold; direction = global invert.
 870|             (feat.rsi_2 < thr, dir_side)
 871|         }
