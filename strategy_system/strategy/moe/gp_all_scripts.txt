"""Leakage-aware feature and model evaluation for offline GP research."""
from __future__ import annotations

from tqdm.auto import tqdm
import numpy as np
import pandas as pd


def _matrix(X):
    if isinstance(X, pd.DataFrame):
        return X.to_numpy(dtype=float), list(X.columns)
    array = np.asarray(X, dtype=float)
    return array, [f"x{i}" for i in range(array.shape[1])]


def _impute_scale(train, test):
    med = np.nanmedian(np.where(np.isfinite(train), train, np.nan), axis=0)
    med = np.where(np.isfinite(med), med, 0.0)
    train = np.where(np.isfinite(train), train, med)
    test = np.where(np.isfinite(test), test, med)
    mean = train.mean(axis=0)
    scale = train.std(axis=0)
    scale[scale < 1e-12] = 1.0
    return (train - mean) / scale, (test - mean) / scale


def canonical_model_name(model_type):
    key = "".join(ch for ch in str(model_type).casefold() if ch.isalnum())
    aliases = {"ridge": "ridge", "l2": "ridge", "ols": "ols", "linear": "ols",
               "ordinaryleastsquares": "ols", "tree": "tree", "decisiontree": "tree",
               "neuralnetwork": "neural_network", "neural": "neural_network", "nn": "neural_network", "mlp": "neural_network"}
    if key not in aliases:
        raise ValueError(f"unknown predictive model {model_type!r}; choose Ridge, OLS, Tree, or Neural Network")
    return aliases[key]


def validate_evaluation_models(model_types):
    if not model_types:
        raise ValueError("evaluation_models must contain at least one model")
    return tuple(dict.fromkeys(canonical_model_name(name) for name in model_types))


def fit_predictive_model(model_type, X_train, y_train, X_test, model_params=None):
    """Fit one interchangeable model and return predictions; uses numpy only."""
    params = model_params or {}
    train, _ = _matrix(X_train)
    test, _ = _matrix(X_test)
    y = np.asarray(y_train, dtype=float).reshape(-1)
    valid = np.isfinite(y) & np.isfinite(train).all(axis=1)
    train, y = train[valid], y[valid]
    if not len(y):
        raise ValueError("no finite training samples")
    train, test = _impute_scale(train, test)
    center = y.mean()
    yc = y - center
    name = canonical_model_name(model_type)
    if name == "ridge":
        alpha = float(params.get("alpha", 1.0))
        coef = np.linalg.solve(train.T @ train + alpha * np.eye(train.shape[1]), train.T @ yc)
        return test @ coef + center
    if name == "ols":
        coef = np.linalg.lstsq(np.column_stack([np.ones(len(train)), train]), y, rcond=None)[0]
        return np.column_stack([np.ones(len(test)), test]) @ coef
    if name == "tree":
        # Small deterministic regression tree, recursively splitting by squared error.
        def grow(tx, ty, vx, depth):
            if depth <= 0 or len(ty) < int(params.get("min_samples_leaf", 8)) * 2:
                return np.full(len(vx), float(ty.mean()))
            best = None
            for col in range(tx.shape[1]):
                threshold = np.median(tx[:, col])
                left = tx[:, col] <= threshold
                if left.sum() < 2 or (~left).sum() < 2:
                    continue
                loss = ((ty[left] - ty[left].mean()) ** 2).sum() + ((ty[~left] - ty[~left].mean()) ** 2).sum()
                if best is None or loss < best[0]:
                    best = (loss, col, threshold, left)
            if best is None:
                return np.full(len(vx), float(ty.mean()))
            _, col, threshold, left = best
            out = np.empty(len(vx))
            mask = vx[:, col] <= threshold
            out[mask] = grow(tx[left], ty[left], vx[mask], depth - 1)
            out[~mask] = grow(tx[~left], ty[~left], vx[~mask], depth - 1)
            return out
        return grow(train, y, test, int(params.get("max_depth", 3)))
    if name == "neural_network":
        rng = np.random.default_rng(int(params.get("seed", 0)))
        hidden = int(params.get("hidden_units", 16))
        w1 = rng.normal(0, 1 / np.sqrt(train.shape[1]), (train.shape[1], hidden))
        b1 = np.zeros(hidden)
        w2 = np.zeros(hidden)
        b2 = 0.0
        lr = float(params.get("learning_rate", 0.01))
        for _ in range(int(params.get("epochs", 120))):
            h = np.tanh(train @ w1 + b1)
            pred = h @ w2 + b2
            err = (pred - yc) / len(y)
            gw2 = h.T @ err
            gb2 = err.sum()
            gh = (err[:, None] * w2) * (1 - h * h)
            w1 -= lr * train.T @ gh
            b1 -= lr * gh.sum(axis=0)
            w2 -= lr * gw2
            b2 -= lr * gb2
        return np.tanh(test @ w1 + b1) @ w2 + b2 + center
    raise AssertionError("canonical_model_name returned an unsupported model")


def prefilter_features(X, y, *, min_abs_correlation=0.0, max_features=None):
    """Cheap univariate Pearson filter; callers must pass training rows only."""
    matrix, names = _matrix(X)
    target = np.asarray(y, dtype=float)
    scores = []
    for j, name in enumerate(names):
        valid = np.isfinite(matrix[:, j]) & np.isfinite(target)
        corr = np.corrcoef(matrix[valid, j], target[valid])[0, 1] if valid.sum() >= 3 else np.nan
        if np.isfinite(corr) and abs(corr) >= min_abs_correlation:
            scores.append((name, abs(float(corr)), j))
    scores.sort(key=lambda row: row[1], reverse=True)
    if max_features is not None:
        scores = scores[:max(0, int(max_features))]
    return [row[0] for row in scores]


def ridge_select_features(X, y, *, alpha=1.0, max_features=None, min_weight=1e-8):
    """Rank features using training-only Ridge coefficients, then remove collinear extras."""
    ranked = [(name, weight) for name, weight in _ridge_importance(X, np.asarray(y, float), alpha)
              if weight >= min_weight]
    matrix, names = _matrix(X)
    selected = []
    for name, _ in ranked:
        idx = names.index(name)
        if selected:
            prior = [names.index(item) for item in selected]
            candidate = matrix[:, idx]
            redundant = False
            for col in prior:
                a, b = matrix[:, col], candidate
                valid = np.isfinite(a) & np.isfinite(b)
                if valid.sum() >= 3 and np.std(a[valid]) > 1e-12 and np.std(b[valid]) > 1e-12:
                    if abs(np.corrcoef(a[valid], b[valid])[0, 1]) >= 0.98:
                        redundant = True
                        break
            if redundant:
                continue
        selected.append(name)
        if max_features is not None and len(selected) >= max_features:
            break
    return selected


def _ridge_importance(X, y, alpha):
    x, names = _matrix(X)
    valid = np.isfinite(y) & np.isfinite(x).all(axis=1)
    x = x[valid]
    target = np.asarray(y, float)[valid]
    if not len(target):
        return []
    x, _ = _impute_scale(x, x)
    yy = target - target.mean()
    coef = np.linalg.solve(x.T @ x + alpha * np.eye(x.shape[1]), x.T @ yy)
    return sorted(zip(names, np.abs(coef)), key=lambda item: item[1], reverse=True)

def run_purged_walk_forward(X, y, *, model_type="ridge", model_params=None, folds=4, purge_rows=0,
                            min_samples=30, regime_labels=None, regime_id=None, episode_ids=None):
    """Expanding chronological folds, with embargo, optionally scored on one latent only."""
    canonical_model = canonical_model_name(model_type)
    matrix, names = _matrix(X)
    target = np.asarray(y, dtype=float)
    if len(matrix) != len(target):
        raise ValueError("X and y row counts must match")
    eligible = np.isfinite(target) & np.isfinite(matrix).any(axis=1)
    if regime_labels is not None:
        labels = np.asarray(regime_labels, dtype=object)
        normalized = np.asarray([str(x).casefold() if pd.notna(x) else "null" for x in labels], dtype=object)
        if regime_id is not None:
            eligible &= normalized == str(regime_id).casefold()
        else:
            eligible &= pd.notna(labels) & (normalized != "null")
        if episode_ids is None:
            episode_ids = np.cumsum(np.r_[True, normalized[1:] != normalized[:-1]])
    if episode_ids is not None:
        episode_ids = np.asarray(episode_ids)
        if len(episode_ids) != len(target):
            raise ValueError("episode_ids must align with X and y")
    positions = np.flatnonzero(eligible)
    if len(positions) < min_samples:
        return {"model": canonical_model, "folds": [], "n_samples": len(positions), "prediction_error": np.nan,
                "directional_accuracy": np.nan, "expectancy": np.nan, "stability": np.nan}
    if episode_ids is not None:
        episode_order = list(dict.fromkeys(episode_ids[positions].tolist()))
        # Hold out individual recurring episodes. Use the latest folds while
        # every earlier episode remains available to the expanding train set.
        test_episodes = episode_order[1:][-max(1, int(folds)):]
        episode_chunks = [np.asarray([episode], dtype=object) for episode in test_episodes]
        splits = []
        episode_ordinal = {episode: i for i, episode in enumerate(episode_order)}
        for chunk in episode_chunks:
            if len(chunk):
                first = episode_ordinal[chunk[0]]
                train_pos = positions[np.asarray([episode_ordinal[e] < first for e in episode_ids[positions]])]
                test_pos = positions[np.isin(episode_ids[positions], chunk)]
                splits.append((train_pos, test_pos, chunk[0]))
    else:
        chunks = np.array_split(positions, max(2, int(folds) + 1))[1:]
        splits = []
        for chunk in chunks:
            if len(chunk):
                start = int(chunk[0])
                train_pos = positions[positions < max(0, start - int(purge_rows))]
                splits.append((train_pos, chunk, None))
    fold_metrics = []
    for train_pos, test_pos, episode_id in tqdm(splits, desc="Evaluating Folds", leave=False):
        train_pos = train_pos[train_pos < max(0, int(test_pos[0]) - int(purge_rows))]
        if len(train_pos) < min_samples or len(test_pos) == 0:
            continue
        pred = fit_predictive_model(canonical_model, matrix[train_pos], target[train_pos], matrix[test_pos], model_params)
        actual = target[test_pos]
        error = float(np.sqrt(np.mean((actual - pred) ** 2)))
        train_mean = float(np.mean(target[train_pos]))
        fold_target_scale = float(np.std(target[train_pos]))
        signal = np.sign(pred - train_mean)
        centered_actual = actual - train_mean
        direction = float(np.mean(signal == np.sign(centered_actual)))
        expectancy = float(np.mean(signal * centered_actual))
        fold_metrics.append({"n_train": len(train_pos), "n_test": len(test_pos), "episode_id": episode_id,
                             "prediction_error": error, "directional_accuracy": direction,
                             "expectancy": expectancy, "target_scale": fold_target_scale})
    if not fold_metrics:
        return {"model": canonical_model, "folds": [], "n_samples": len(positions), "prediction_error": np.nan,
                "directional_accuracy": np.nan, "expectancy": np.nan, "stability": np.nan}
    errors = np.array([f["prediction_error"] for f in fold_metrics])
    acc = np.array([f["directional_accuracy"] for f in fold_metrics])
    exp = np.array([f["expectancy"] for f in fold_metrics])
    target_scale = float(np.nanmean([f["target_scale"] for f in fold_metrics]))
    stability = float(np.clip(1.0 - np.std(exp) / max(target_scale, 1e-12), 0.0, 1.0))
    signs = np.sign(exp[np.abs(exp) > 1e-12])
    recurrence_consistency = float(max(np.mean(signs > 0), np.mean(signs < 0))) if signs.size else np.nan
    return {"model": canonical_model, "folds": fold_metrics, "n_samples": len(positions),
            "prediction_error": float(np.mean(errors)), "directional_accuracy": float(np.mean(acc)),
            "expectancy": float(np.mean(exp)), "stability": stability,
            "expectancy_std": float(np.std(exp)), "target_scale": target_scale,
            "episode_expectancy_std": float(np.std(exp)), "recurrence_consistency": recurrence_consistency,
            "episode_count": len({str(f["episode_id"]) for f in fold_metrics if f["episode_id"] is not None}),
            "feature_names": names}


def calculate_feature_stability(fold_scores, target_scale=None):
    values = np.asarray(fold_scores, dtype=float)
    values = values[np.isfinite(values)]
    if not values.size or target_scale is None or not np.isfinite(target_scale):
        return np.nan
    return float(np.clip(1.0 - np.std(values) / max(float(target_scale), 1e-12), 0.0, 1.0))


def calculate_model_stability(results):
    scales = np.asarray([row.get("target_scale", np.nan) for row in results], dtype=float)
    scale = float(np.nanmean(scales)) if np.isfinite(scales).any() else None
    episode_scores = [fold.get("expectancy", np.nan) for row in results for fold in row.get("folds", ())]
    scores = episode_scores or [row.get("expectancy", np.nan) for row in results]
    return calculate_feature_stability(scores, scale)


def calculate_prediction_error(y_true, y_pred):
    return float(np.sqrt(np.nanmean((np.asarray(y_true, float) - np.asarray(y_pred, float)) ** 2)))


def calculate_directional_accuracy(y_true, y_pred, train_mean=None):
    baseline = float(np.nanmean(y_true)) if train_mean is None else float(train_mean)
    return float(np.nanmean(np.sign(np.asarray(y_pred) - baseline) == np.sign(np.asarray(y_true) - baseline)))


def calculate_expectancy(y_true, y_pred, train_mean=None):
    actual = np.asarray(y_true, float)
    baseline = float(np.nanmean(actual)) if train_mean is None else float(train_mean)
    return float(np.nanmean(np.sign(np.asarray(y_pred) - baseline) * (actual - baseline)))


def calculate_complexity_penalty(complexity, coefficient=0.0005):
    return float(coefficient * np.log1p(max(0, complexity)))


def calculate_fitness(result, *, complexity=1, complexity_penalty=0.0005):
    expectancy = result.get("expectancy", np.nan)
    stability = result.get("stability", np.nan)
    error = result.get("prediction_error", np.nan)
    if not np.isfinite(expectancy) or not np.isfinite(stability) or not np.isfinite(error):
        return -np.inf
    scale = result.get("target_scale", np.nan)
    if not np.isfinite(scale) or scale <= 0:
        return -np.inf
    return float((expectancy / scale) * stability - 0.05 * (error / scale)
                 - calculate_complexity_penalty(complexity, complexity_penalty))

_FORBIDDEN_NAME_PARTS = ("future_", "target", "label", "barrier", "latent", "trade_allowed", "embedding")


def evaluate_population(population, evaluator):
    """Evaluate an expression population through a caller-provided objective."""
    return [(expression, float(evaluator(expression))) for expression in population]


def select_parents(rng, scored_population, count=2, tournament_size=4):
    return [tournament_select(rng, scored_population, tournament_size) for _ in range(count)]


def evolve_generation(rng, scored_population, primitive_features, *, population_size, max_depth,
                     temporal_windows=(2, 4, 8, 16), elite_count=2, tournament_size=4,
                     crossover_rate=0.7, mutation_rate=0.3, immigrant_fraction=0.0):
    """Create one bounded next generation from a scored expression population."""
    scored = sorted(scored_population, key=lambda item: item[1], reverse=True)
    next_generation = [node for node, _ in scored[:max(0, elite_count)]]
    immigrant_count = int(round(population_size * immigrant_fraction))
    target = population_size - immigrant_count
    while len(next_generation) < target:
        left = select_parents(rng, scored, 1, tournament_size)[0]
        if rng.random() < crossover_rate:
            right = select_parents(rng, scored, 1, tournament_size)[0]
            child = crossover(rng, left, right, max_depth)
        else:
            child = left
        if rng.random() < mutation_rate:
            child = mutate(rng, child, tuple(primitive_features), max_depth, tuple(temporal_windows))
        next_generation.append(child)
    while len(next_generation) < population_size:
        next_generation.append(random_tree(rng, tuple(primitive_features), max_depth,
                                           temporal_windows=tuple(temporal_windows), grow=True))
    return next_generation


def stop_evolution(best_scores, *, patience=3, min_improvement=1e-6):
    """Stop after a stable fitness plateau or a non-finite score sequence."""
    if len(best_scores) <= patience:
        return False
    recent = np.asarray(best_scores[-(patience + 1):], dtype=float)
    if not np.isfinite(recent[-1]):
        return True
    return bool(np.nanmax(recent[1:]) - np.nanmax(recent[:-1]) <= min_improvement)


def evolve_population(rng, primitive_features, evaluator, *, population_size=32, max_depth=5,
                      generations=8, temporal_windows=(2, 4, 8, 16), initial_population=None,
                      population_filter=None, patience=3, **generation_kwargs):
    """Generic offline GP loop; evaluator and optional screening control the objective."""
    population = list(initial_population) if initial_population is not None else create_initial_population(
        rng, primitive_features, population_size, max_depth, temporal_windows)
    history = []
    hall_of_fame = {}
    for gen in tqdm(range(generations), desc="Evolving Generations", leave=False):
        if population_filter is not None:
            population = population_filter(population)
        while len(population) < population_size:
            population.append(random_tree(rng, tuple(primitive_features), max_depth,
                                          temporal_windows=tuple(temporal_windows), grow=True))
        scored = evaluate_population(population, evaluator)
        for expression, score in scored:
            hall_of_fame[str(expression)] = (expression, score)
        history.append(max((score for _, score in scored), default=-np.inf))
        if gen == generations - 1 or stop_evolution(history, patience=patience):
            break
        population = evolve_generation(rng, scored, primitive_features, population_size=population_size,
                                       max_depth=max_depth, temporal_windows=temporal_windows,
                                       **generation_kwargs)
    return sorted(hall_of_fame.values(), key=lambda item: item[1], reverse=True)

def _is_price_level_name(name: str) -> bool:
    """Recognize raw price-level names, including bid/ask and namespaced fields."""
    normalized = str(name).lower()
    parts = normalized.replace("-", "_").split("_")
    if any(part in {
        "return", "returns", "ret", "change", "pct", "percent", "qty", "quantity",
        "size", "volume", "depth", "imbalance", "flow", "count", "ratio",
    } for part in parts):
        return False
    if {"price", "close", "mid", "last"}.intersection(parts):
        return True
    return bool(parts and parts[-1] in {"bid", "ask"})


def build_gp_inputs(
    frame: pd.DataFrame,
    input_columns: tuple[str, ...] | list[str] | None = None,
    normalization_window: int = 500,
    normalization_min_periods: int = 50,
    clip_z: float = 8.0,
) -> pd.DataFrame:
    """Create causal, scale-normalized GP terminals from available features.

    Price levels are converted to log returns. Other numeric columns are
    standardized against preceding rows only. Target-like and registry columns
    are excluded by default; explicit columns are still checked for leakage.
    """
    if not isinstance(frame, pd.DataFrame) or frame.empty:
        raise ValueError("frame must be a nonempty DataFrame")
    if normalization_window < 2 or not 2 <= normalization_min_periods <= normalization_window:
        raise ValueError("normalization_min_periods must be in [2, normalization_window]")
    if not np.isfinite(clip_z) or clip_z <= 0:
        raise ValueError("clip_z must be finite and > 0")

    if input_columns is None:
        columns = [
            column for column in frame.columns
            if pd.api.types.is_numeric_dtype(frame[column])
            and not any(part in str(column).lower() for part in _FORBIDDEN_NAME_PARTS)
            and "timestamp" not in str(column).lower()
            and not str(column).lower().endswith("_id")
        ]
    else:
        columns = list(input_columns)
        missing = [column for column in columns if column not in frame.columns]
        if missing:
            raise KeyError(f"Missing GP input columns: {missing}")
        forbidden = [column for column in columns if any(part in str(column).lower() for part in _FORBIDDEN_NAME_PARTS)]
        if forbidden:
            raise ValueError(f"target/label/state columns cannot be GP inputs: {forbidden}")
    if not columns:
        raise ValueError("no numeric causal GP input columns were found")

    output: dict[str, pd.Series] = {}
    for column in columns:
        values = pd.to_numeric(frame[column], errors="coerce").astype(np.float64)
        if _is_price_level_name(str(column)):
            logged = np.log(values.where(values > 0.0))
            values = logged.diff()
            input_name = f"x__log_return__{column}"
        else:
            input_name = f"x__{column}"
        prior = values.shift(1)
        history_mean = prior.rolling(normalization_window, min_periods=normalization_min_periods).mean()
        history_std = prior.rolling(normalization_window, min_periods=normalization_min_periods).std(ddof=0)
        delta = values - history_mean
        zscore = delta.div(history_std.where(history_std > 1e-12))
        flat = history_std.le(1e-12) & history_mean.notna()
        zscore = zscore.mask(flat & delta.abs().le(1e-12), 0.0)
        zscore = zscore.mask(flat & delta.gt(1e-12), clip_z)
        zscore = zscore.mask(flat & delta.lt(-1e-12), -clip_z)
        output[input_name] = zscore.clip(-clip_z, clip_z)

    result = pd.DataFrame(output, index=frame.index).replace([np.inf, -np.inf], np.nan)
    result = result.loc[:, result.notna().any(axis=0)]
    if result.empty:
        raise ValueError("GP inputs contain no usable values after causal normalization")
    return result

@dataclass
class GeneticFeatureModel:
    """Selected state-specific expression trees and their input transform."""

    selected: tuple[SelectedExpression, ...]
    input_columns: tuple[str, ...] | None
    normalization_window: int
    normalization_min_periods: int
    clip_z: float

    def transform(self, frame: pd.DataFrame, state_labels: pd.Series) -> pd.DataFrame:
        """Evaluate selected features causally and mask them to their state."""
        if not frame.index.equals(state_labels.index):
            raise ValueError("frame and state_labels must have identical indexes")
        inputs = build_gp_inputs(
            frame,
            input_columns=self.input_columns,
            normalization_window=self.normalization_window,
            normalization_min_periods=self.normalization_min_periods,
            clip_z=self.clip_z,
        )
        state_values = state_labels.astype("object").map(
            lambda value: "null" if pd.isna(value) else str(value).casefold())
        output = pd.DataFrame(index=frame.index)
        for item in self.selected:
            values = evaluate_expression(item.expression, inputs)
            output[item.output_column] = values.where(state_values == item.state_id.casefold())
        return output

    def to_dict(self) -> dict[str, Any]:
        """Serialize the fitted expression inventory without executable code."""
        return {
            "input_columns": list(self.input_columns) if self.input_columns is not None else None,
            "normalization_window": self.normalization_window,
            "normalization_min_periods": self.normalization_min_periods,
            "clip_z": self.clip_z,
            "selected": [
                {
                    "state_id": item.state_id,
                    "target_column": item.target_column,
                    "output_column": item.output_column,
                    "expression": _node_to_dict(item.expression),
                    "train_ic": item.train_ic,
                    "validation_ic": item.validation_ic,
                    "fitness": item.fitness,
                    "direction": item.direction,
                    "n_train": item.n_train,
                    "n_validation": item.n_validation,
                    "model_metrics": item.model_metrics,
                }
                for item in self.selected
            ],
        }

    @classmethod
    def from_dict(cls, state: dict[str, Any]) -> "GeneticFeatureModel":
        selected = tuple(
            SelectedExpression(
                state_id=item["state_id"],
                target_column=item["target_column"],
                output_column=item["output_column"],
                expression=_node_from_dict(item["expression"]),
                train_ic=float(item["train_ic"]),
                validation_ic=float(item["validation_ic"]),
                fitness=float(item["fitness"]),
                n_train=int(item["n_train"]),
                n_validation=int(item["n_validation"]),
                direction=int(item.get("direction", np.sign(item["train_ic"]))),
                model_metrics=dict(item.get("model_metrics", {})),
            )
            for item in state["selected"]
        )
        return cls(
            selected=selected,
            input_columns=tuple(state["input_columns"]) if state.get("input_columns") is not None else None,
            normalization_window=int(state["normalization_window"]),
            normalization_min_periods=int(state["normalization_min_periods"]),
            clip_z=float(state["clip_z"]),
        )

class GPEvolutionResult:
    features: pd.DataFrame
    report: pd.DataFrame
    model: GeneticFeatureModel
    feature_library: dict[str, Any] | None = None


def _spearman(left: np.ndarray, right: np.ndarray, min_samples: int) -> tuple[float, int]:
    """Tie-averaged Spearman rho via the numba kernel (no per-call pandas rank allocs)."""
    a = np.ascontiguousarray(left, dtype=np.float64).reshape(-1)
    b = np.ascontiguousarray(right, dtype=np.float64).reshape(-1)
    if a.size != b.size:
        raise ValueError("spearman inputs must have equal length")
    rho, count = _numba_spearman(a, b)
    count = int(count)
    if count < min_samples:
        return np.nan, count
    return float(rho), count


def _correlation_tstat(correlation: float, sample_count: int) -> float:
    if sample_count <= 2 or not np.isfinite(correlation):
        return np.nan
    bounded = min(abs(float(correlation)), 1.0 - 1e-12)
    return bounded * np.sqrt((sample_count - 2) / max(1e-12, 1.0 - bounded * bounded))


def _multiple_testing_tstat_threshold(test_count: int, alpha: float) -> float:
    """Conservative normal-tail Bonferroni threshold for a GP search family."""
    return float(np.sqrt(2.0 * np.log(max(2.0, 2.0 * max(1, test_count) / alpha))))


def _validate_train_mask(train_mask: pd.Series | np.ndarray, index: pd.Index) -> np.ndarray:
    if isinstance(train_mask, pd.Series):
        if not train_mask.index.equals(index):
            raise ValueError("train_mask Series must have the same index as features")
        mask = train_mask.fillna(False).to_numpy(dtype=bool)
    else:
        mask = np.asarray(train_mask, dtype=bool)
        if mask.ndim != 1 or len(mask) != len(index):
            raise ValueError("train_mask must be one-dimensional and match the rows")
    seen_validation = np.maximum.accumulate(~mask)
    if np.any(mask & seen_validation):
        raise ValueError("train_mask must be a chronological prefix")
    if not mask.any() or mask.all():
        raise ValueError("train_mask must contain both training and validation rows")
    return mask


def _purge_target_overlap(
    train: np.ndarray,
    features: pd.DataFrame,
    targets: pd.DataFrame,
    target_columns: tuple[str, ...],
    train_end_time=None,
    purge_rows: int | None = None,
) -> np.ndarray:
    """Remove train rows whose forward target can reach the validation period."""
    if purge_rows is not None and (not isinstance(purge_rows, (int, np.integer)) or purge_rows < 0):
        raise ValueError("purge_rows must be an integer >= 0 or None")
    validation_positions = np.flatnonzero(~train)
    if not validation_positions.size:
        raise ValueError("train_mask must leave validation rows after the training prefix")
    split = int(validation_positions[0])
    cutoff = train_end_time
    if cutoff is None and isinstance(features.index, pd.DatetimeIndex):
        if features.index.isna().any() or not features.index.is_monotonic_increasing:
            raise ValueError("DatetimeIndex must be valid and increasing for target purging")
        cutoff = features.index[split]
    result = train.copy()
    if cutoff is not None:
        cutoff = pd.to_datetime(cutoff, utc=True)
        needs_row_purge = False
        endpoint_columns = [
            f"future_timestamp__{column.rsplit('__', 1)[-1]}"
            if column.startswith(("future_return__", "future_mfe_", "future_mae_")) else None
            for column in target_columns
        ]
        for target_column, endpoint_column in zip(target_columns, endpoint_columns):
            if endpoint_column is not None and endpoint_column in targets.columns:
                endpoints = pd.to_datetime(targets[endpoint_column], utc=True, errors="coerce")
                result &= np.asarray(endpoints < cutoff)
                continue
            horizon_label = target_column.rsplit("__", 1)[-1]
            if isinstance(features.index, pd.DatetimeIndex):
                if horizon_label.endswith("bar"):
                    try:
                        bars = int(horizon_label[:-3])
                    except ValueError:
                        bars = 0
                    positions = np.arange(len(features))
                    result &= (positions + bars < split) if bars > 0 else False
                else:
                    try:
                        horizon = pd.to_timedelta(horizon_label)
                    except (TypeError, ValueError):
                        horizon = pd.NaT
                    if pd.isna(horizon) or horizon <= pd.Timedelta(0):
                        if purge_rows is None:
                            raise ValueError(f"cannot infer target horizon from {target_column!r}; provide future timestamps or purge_rows")
                        result[max(0, split - int(purge_rows)):split] = False
                        continue
                    times = pd.to_datetime(features.index, utc=True, errors="coerce")
                    result &= np.asarray(times + horizon < cutoff)
            elif purge_rows is None:
                needs_row_purge = True
            else:
                needs_row_purge = True
        if isinstance(features.index, pd.DatetimeIndex):
            feature_times = pd.to_datetime(features.index, utc=True, errors="coerce")
            result &= np.asarray(feature_times < cutoff)
        elif needs_row_purge:
            row_horizons = []
            for target_column in target_columns:
                label = target_column.rsplit("__", 1)[-1]
                if label.endswith("bar"):
                    try:
                        row_horizons.append(int(label[:-3]))
                    except ValueError:
                        pass
            embargo = int(purge_rows or 0) or (max(row_horizons) if row_horizons else 0)
            if embargo <= 0:
                raise ValueError(
                    "cannot purge target overlap without timestamp endpoints; provide purge_rows"
                )
            result[max(0, split - embargo):split] = False
        return result

    if purge_rows is None or purge_rows == 0:
        # Integer bar horizons are enough to derive a conservative embargo.
        bar_horizons = []
        for target_column in target_columns:
            label = target_column.rsplit("__", 1)[-1]
            if label.endswith("bar"):
                try:
                    bar_horizons.append(int(label[:-3]))
                except ValueError:
                    pass
        if bar_horizons:
            purge_rows = max(bar_horizons)
        else:
            raise ValueError("cannot purge target overlap; provide target timestamps, a DatetimeIndex, or purge_rows")
    if purge_rows:
        result[max(0, split - int(purge_rows)):split] = False
    return result


def _infer_purge_rows(target_column: str, index: pd.Index) -> int:
    """Infer an inner-fold embargo from the encoded target horizon when possible."""
    label = str(target_column).rsplit("__", 1)[-1].lower()
    if label.endswith("bar"):
        try:
            return max(1, int(label[:-3]))
        except ValueError:
            pass
    if isinstance(index, pd.DatetimeIndex) and len(index) > 2:
        try:
            horizon = pd.to_timedelta(label)
            cadence = pd.Series(index).diff().dropna().median()
            if horizon > pd.Timedelta(0) and cadence > pd.Timedelta(0):
                return max(1, int(np.ceil(horizon / cadence)))
        except (TypeError, ValueError, ZeroDivisionError):
            pass
    return 1


def _blockwise_spearman(
    values: np.ndarray,
    target: np.ndarray,
    train_mask: np.ndarray,
    train_end: int,
    block_count: int,
    min_block_samples: int,
) -> np.ndarray:
    """Compute separate train ICs over chronological blocks (one numba kernel)."""
    v = np.ascontiguousarray(values, dtype=np.float64).reshape(-1)
    t = np.ascontiguousarray(target, dtype=np.float64).reshape(-1)
    m = np.ascontiguousarray(train_mask, dtype=np.bool_).reshape(-1)
    if not (v.size == t.size == m.size) or train_end > v.size:
        raise ValueError("blockwise spearman inputs must share one length and cover train_end")
    return np.asarray(
        _numba_blockwise(v, t, m, int(train_end), int(block_count), int(min_block_samples)),
        dtype=np.float64,
    )


def _is_correlated_duplicate(
    candidate: np.ndarray,
    selected: list[np.ndarray],
    mask: np.ndarray,
    threshold: float,
    min_samples: int,
) -> bool:
    from .jit_kernels import abs_corr_exceeds as _exceeds

    cand = np.ascontiguousarray(np.asarray(candidate, dtype=np.float64).reshape(-1))
    mk = np.ascontiguousarray(np.asarray(mask, dtype=np.bool_).reshape(-1))
    for prior in selected:
        pr = np.ascontiguousarray(np.asarray(prior, dtype=np.float64).reshape(-1))
        if bool(_exceeds(cand, pr, mk, float(threshold), int(min_samples))):
            return True
    return False


def extrapolate_features(
    features: pd.DataFrame,
    targets: pd.DataFrame,
    state_labels: pd.Series,
    train_mask: pd.Series | np.ndarray,
    *,
    target_columns: tuple[str, ...] | list[str] | None = None,
    input_columns: tuple[str, ...] | list[str] | None = None,
    normalization_window: int = 500,
    normalization_min_periods: int = 50,
    clip_z: float = 8.0,
    max_height: int = 5,
    population_size: int = 32,
    generations: int = 8,
    tournament_size: int = 4,
    elite_count: int = 2,
    mutation_rate: float = 0.30,
    crossover_rate: float = 0.70,
    max_features_per_state_target: int = 2,
    min_state_samples: int = 60,
    include_null: bool = False,
    min_null_samples: int = 60,
    max_states: int = 10,
    temporal_windows: tuple[int, ...] | list[int] = (2, 4, 8, 16),
    complexity_penalty: float = 0.0005,
    block_count: int = 5,
    min_block_samples: int = 20,
    min_valid_blocks: int = 3,
    require_block_sign_agreement: bool = True,
    block_std_penalty: float = 0.5,
    min_train_ic: float = 0.03,
    min_validation_ic: float = 0.02,
    min_train_tstat: float = 2.0,
    min_validation_tstat: float = 1.96,
    multiple_testing_alpha: float = 0.05,
    validation_ic_retention: float = 0.25,
    correlation_threshold: float = 0.98,
    immigrant_fraction: float = 0.10,
    train_end_time=None,
    purge_rows: int | None = None,
    seed: int = 7,
    evaluation_models: tuple[str, ...] | list[str] = ("ridge", "ols", "tree", "neural_network"),
    evaluation_model_params: dict[str, dict[str, Any]] | None = None,
    feature_library_path: str | None = None,
) -> GPEvolutionResult:
    """Evolve causal features per known latent state and target.

    Fitness uses mean IC across chronological training blocks, penalizes
    instability and tree size, and requires block sign agreement by default.
    A held-out validation veto rejects weak or sign-flipped features but never
    ranks candidates. Forward-target overlap is purged at the train boundary.
    Unknown/null states are excluded.
    """
    if not features.index.equals(targets.index) or not features.index.equals(state_labels.index):
        raise ValueError("features, targets, and state_labels must have identical indexes")
    if isinstance(features.index, pd.DatetimeIndex) and (
        features.index.isna().any() or not features.index.is_monotonic_increasing
    ):
        raise ValueError("DatetimeIndex must be valid and increasing for chronological GP validation")
    if not isinstance(max_height, int) or max_height < 2:
        raise ValueError("max_height must be an integer >= 2")
    if not isinstance(population_size, int) or population_size < 4:
        raise ValueError("population_size must be an integer >= 4")
    if not isinstance(generations, int) or generations < 1:
        raise ValueError("generations must be an integer >= 1")
    if not 1 <= elite_count < population_size:
        raise ValueError("elite_count must be in [1, population_size - 1]")
    if not 0.0 <= mutation_rate <= 1.0 or not 0.0 <= crossover_rate <= 1.0:
        raise ValueError("mutation_rate and crossover_rate must be in [0, 1]")
    if max_features_per_state_target < 1 or min_state_samples < 2 or max_states < 1:
        raise ValueError("feature, sample, and state limits must be positive")
    if min_null_samples < 2:
        raise ValueError("min_null_samples must be >= 2")
    if not np.isfinite(complexity_penalty) or complexity_penalty < 0.0:
        raise ValueError("complexity_penalty must be finite and >= 0")
    if not isinstance(block_count, int) or block_count < 2:
        raise ValueError("block_count must be an integer >= 2")
    if not isinstance(min_block_samples, int) or min_block_samples < 3:
        raise ValueError("min_block_samples must be an integer >= 3")
    if not isinstance(min_valid_blocks, int) or not 1 <= min_valid_blocks <= block_count:
        raise ValueError("min_valid_blocks must be in [1, block_count]")
    if not np.isfinite(block_std_penalty) or block_std_penalty < 0.0:
        raise ValueError("block_std_penalty must be finite and >= 0")
    if not np.isfinite(min_train_ic) or not 0.0 <= min_train_ic <= 1.0:
        raise ValueError("min_train_ic must be in [0, 1]")
    if not np.isfinite(min_validation_ic) or not 0.0 <= min_validation_ic <= 1.0:
        raise ValueError("min_validation_ic must be in [0, 1]")
    if not np.isfinite(min_train_tstat) or min_train_tstat < 0:
        raise ValueError("min_train_tstat must be finite and >= 0")
    if not np.isfinite(min_validation_tstat) or min_validation_tstat < 0:
        raise ValueError("min_validation_tstat must be finite and >= 0")
    if not np.isfinite(multiple_testing_alpha) or not 0.0 < multiple_testing_alpha < 1.0:
        raise ValueError("multiple_testing_alpha must be in (0, 1)")
    if not np.isfinite(validation_ic_retention) or not 0.0 <= validation_ic_retention <= 1.0:
        raise ValueError("validation_ic_retention must be in [0, 1]")
    if not np.isfinite(correlation_threshold) or not 0.0 < correlation_threshold <= 1.0:
        raise ValueError("correlation_threshold must be in (0, 1]")
    if not np.isfinite(immigrant_fraction) or not 0.0 <= immigrant_fraction < 0.5:
        raise ValueError("immigrant_fraction must be in [0, 0.5)")
    evaluation_models = validate_evaluation_models(evaluation_models)
    model_params_by_name = {
        validate_evaluation_models([name])[0]: params
        for name, params in (evaluation_model_params or {}).items()
    }

    inputs = build_gp_inputs(
        features,
        input_columns=input_columns,
        normalization_window=normalization_window,
        normalization_min_periods=normalization_min_periods,
        clip_z=clip_z,
    )
    variables = tuple(inputs.columns)
    initial_train = _validate_train_mask(train_mask, features.index)
    targets_to_use = tuple(
        target_columns
        if target_columns is not None
        else [column for column in targets if column.startswith("future_return__")]
    )
    if not targets_to_use:
        raise ValueError("target_columns must select at least one target column")
    missing_targets = [column for column in targets_to_use if column not in targets]
    if missing_targets:
        raise KeyError(f"GP target columns missing or empty: {missing_targets}")
    split_positions = np.flatnonzero(~initial_train)
    split_boundary = int(split_positions[0])
    if train_end_time is not None and isinstance(features.index, pd.DatetimeIndex):
        ordered_times = pd.to_datetime(features.index, utc=True, errors="coerce")
        split_boundary = int(np.searchsorted(ordered_times.asi8, pd.to_datetime(train_end_time, utc=True).value, side="left"))
    train = _purge_target_overlap(
        initial_train,
        features,
        targets,
        targets_to_use,
        train_end_time=train_end_time,
        purge_rows=purge_rows,
    )
    if not train.any():
        raise ValueError("target-overlap purge removed all GP training rows")
    validation = ~initial_train
    state_values = state_labels.astype("object").to_numpy()
    state_keys = np.asarray([str(value).casefold() if pd.notna(value) else "null" for value in state_values], dtype=object)
    known_state_mask = pd.notna(state_values) & (state_keys != "null") & (state_keys != "")
    state_display = {}
    for value, key, known in zip(state_values, state_keys, known_state_mask):
        if known:
            state_display.setdefault(key, str(value))
    allowed_state_keys = list(state_display)
    if include_null and np.any((state_keys == "null") & train):
        state_display["null"] = "null"
        allowed_state_keys.append("null")
    state_counts = {
        key: int(np.sum(pd.notna(state_values) & (state_keys == key) & train))
        for key in allowed_state_keys
    }
    states = [state_display[key] for key, _ in sorted(state_counts.items(), key=lambda pair: (-pair[1], pair[0]))[:max_states]]
    if not states:
        raise ValueError("no known latent states are available for GP search")

    windows = tuple(int(window) for window in temporal_windows)
    if not windows or any(window < 1 for window in windows):
        raise ValueError("temporal_windows must contain positive integers")
    rng = np.random.default_rng(seed)
    search_tstat = _multiple_testing_tstat_threshold(population_size * generations, multiple_testing_alpha)
    required_train_tstat = max(min_train_tstat, search_tstat)
    required_validation_tstat = max(min_validation_tstat, search_tstat)
    target_values = {column: pd.to_numeric(targets[column], errors="coerce").to_numpy(dtype=np.float64) for column in targets_to_use}
    selected: list[SelectedExpression] = []
    report_rows: list[dict[str, Any]] = []
    train_end = split_boundary
    train_inputs = inputs.iloc[:train_end]
    # One temporal cache for the whole run: (var, window) float32 arrays are
    # precomputed once instead of pandas shift/rolling re-run per candidate.
    # The train prefix is a zero-copy causal view of the same arrays.
    temporal_cache = build_temporal_cache(inputs, windows)
    train_cache = temporal_cache.prefix(train_end)

    # Each state-target pair has an independent population and objective.
    for state_id in tqdm(states, desc="GP State Search"):
        state_mask = state_keys == state_id.casefold()
        for target_column in tqdm(targets_to_use, desc=f"Targets [{state_id}]", leave=False):
            y = target_values[target_column]
            train_mask_pair = train & state_mask & np.isfinite(y)
            validation_mask_pair = validation & state_mask & np.isfinite(y)
            required_samples = min_null_samples if state_id.casefold() == "null" else min_state_samples
            if int(train_mask_pair.sum()) < required_samples:
                continue
            # Cheap target correlation, followed by Ridge coefficient and
            # redundancy filtering, is fit on this latent's training rows only.
            train_rows = np.flatnonzero(train_mask_pair[:train_end])
            primitive_frame = inputs.iloc[train_rows]
            primitive_target = y[train_rows]
            cheap_names = prefilter_features(primitive_frame, primitive_target, max_features=64)
            if not cheap_names:
                continue
            ridge_names = ridge_select_features(primitive_frame[cheap_names], primitive_target,
                                                max_features=min(32, len(cheap_names)))
            search_variables = tuple(ridge_names or cheap_names)
            population = initialize_population(rng, search_variables, population_size, max_height, windows)

            def _train_eval(node):
                """Cached train-prefix eval; pandas fallback keeps invalid-node detection exact."""
                try:
                    return evaluate_expression_cached(node, train_cache, train_inputs.index)
                except (ValueError, KeyError, TypeError, FloatingPointError, AttributeError, OverflowError):
                    return evaluate_expression(node, train_inputs)

            def screen_expression_population(candidates):
                candidates = deduplicate_population(filter_invalid_features(candidates, search_variables, max_height))
                sample_mask = train_mask_pair[:train_end]
                candidates = filter_constant_features(candidates, train_inputs, sample_mask=sample_mask, cache=train_cache)
                if not candidates:
                    return []
                expression_frame = pd.DataFrame(
                    {str(node): _train_eval(node).iloc[train_rows].to_numpy(dtype=float)
                     for node in candidates},
                    index=features.index[train_rows],
                )
                cheap = prefilter_features(expression_frame, primitive_target, max_features=64)
                if not cheap:
                    return []
                ridge = ridge_select_features(expression_frame[cheap], primitive_target,
                                              max_features=min(16, len(cheap)))
                survivors = [node for node in candidates if str(node) in set(ridge)]
                survivors = filter_redundant_features(survivors, train_inputs, threshold=correlation_threshold,
                                                      min_samples=min_block_samples, sample_mask=sample_mask,
                                                      cache=train_cache)
                return survivors

            population = screen_expression_population(population)
            while len(population) < population_size:
                population.append(random_tree(rng, search_variables, max_height, temporal_windows=windows, grow=True))
            final_scored: list[tuple[Node, float, float, int, float, float, int]] = []
            score_cache: dict[str, tuple[float, float, int, float, float, int]] = {}

            def score_expression(node):
                key = str(node)
                if key not in score_cache:
                    try:
                        values = evaluate_expression_cached(node, train_cache, train_inputs.index).to_numpy(dtype=np.float64)
                    except (FloatingPointError, OverflowError, ValueError, KeyError, TypeError, AttributeError):
                        try:
                            values = evaluate_expression(node, train_inputs).to_numpy(dtype=np.float64)
                        except (FloatingPointError, OverflowError, ValueError, KeyError):
                            values = np.full(train_end, np.nan, dtype=np.float64)
                    train_pair = train_mask_pair[:train_end]
                    train_ic, count = _spearman(values[train_pair], y[:train_end][train_pair], min_state_samples)
                    block_ics = _blockwise_spearman(values, y[:train_end], train_pair, train_end,
                                                     block_count, min_block_samples)
                    block_mean = float(block_ics.mean()) if block_ics.size else np.nan
                    block_std = float(block_ics.std(ddof=0)) if block_ics.size else np.nan
                    signs_agree = (block_ics.size > 0 and np.all(np.sign(block_ics) == np.sign(block_ics[0]))
                                   and np.sign(train_ic) == np.sign(block_ics[0]))
                    eligible = (np.isfinite(train_ic)
                                and _correlation_tstat(train_ic, count) >= required_train_tstat
                                and block_ics.size >= min_valid_blocks
                                and abs(block_mean) >= min_train_ic
                                and (signs_agree or not require_block_sign_agreement))
                    objective_scores = []
                    if eligible:
                        masked_y = np.where(train_pair, y[:train_end], np.nan)
                        candidate_frame = pd.DataFrame({"candidate": values}, index=features.index[:train_end])
                        for model_name in evaluation_models:
                            try:
                                wf = test_feature_set(
                                    candidate_frame, masked_y, model_type=model_name,
                                    model_params=model_params_by_name.get(model_name),

                                    min_samples=max(20, required_samples // 5),

                                    regime_id=state_id,
                                )
                                score = calculate_fitness(wf, complexity=node.size,
                                                          complexity_penalty=complexity_penalty)
                                if np.isfinite(score):
                                    objective_scores.append((model_name, score))
                            except (ValueError, KeyError, TypeError, AttributeError,
                                    FloatingPointError, OverflowError, np.linalg.LinAlgError):
                                continue

                    families = dict.fromkeys("linear" if name in {"ridge", "ols"} else name
                                             for name in evaluation_models)
                    for family in families:
                        family_values = [score for name, score in objective_scores
                                         if ("linear" if name in {"ridge", "ols"} else name) == family]
                        if family_values:
                            family_scores.append(float(np.mean(family_values)))
                    fitness = float(np.mean(family_scores)) if family_scores else -np.inf

                return score_cache[key][0]

                (node, score, *score_cache[str(node)][1:])
                for node, score in scored_population

                except (FloatingPointError, OverflowError, ValueError, KeyError, TypeError, AttributeError):

                validation_ic, n_validation = _spearman(
                    values[validation_mask_pair],
                    y[validation_mask_pair],
                    min_state_samples,
                )
                validation_passes = (
                    np.isfinite(validation_ic)
                    and _correlation_tstat(validation_ic, n_validation) >= required_validation_tstat
                    and abs(validation_ic) >= min_validation_ic
                    and abs(validation_ic) >= validation_ic_retention * abs(train_ic)
                    and np.sign(validation_ic) == np.sign(train_ic)
                )
                if not validation_passes:
                    continue
                train_values = values[:train_end]
                if _is_correlated_duplicate(
                    train_values,
                    selected_values,
                    train_mask_pair[:train_end],
                    correlation_threshold,
                    min_state_samples,
                ):
                    continue
                # Compare feature usefulness across model families on the
                # training prefix only. The outer validation block above stays
                # a final veto and never enters these walk-forward folds.

                candidate_frame = pd.DataFrame({"candidate": values[:train_end]}, index=features.index[:train_end])

                                        min_samples=max(20, required_samples // 5),

                        )
                    except (ValueError, KeyError, TypeError, AttributeError,

                        model_metrics[f"{prefix}_recurrence_consistency"] = result.get("recurrence_consistency", np.nan)

                            calculate_fitness(result, complexity=node.size, complexity_penalty=complexity_penalty))
                    fitness = float(np.mean([np.mean(scores) for scores in score_by_family.values()]))
                else:

                item = SelectedExpression(
                    state_id=state_id,
                    target_column=target_column,
                    output_column=output_column,
                    expression=node,
                    train_ic=float(train_ic),
                    validation_ic=float(validation_ic),
                    fitness=float(fitness),
                    n_train=int(n_train),
                    n_validation=int(n_validation),
                    direction=int(np.sign(train_ic)),
                    model_metrics=model_metrics,
                )
                selected.append(item)
                selected_values.append(train_values)
                report_rows.append({
                    "state_id": state_id,
                    "target_column": target_column,
                    "output_column": output_column,

                    "n_train": int(n_train),

            selected[i].output_column: evaluate_expression(selected[i].expression, inputs).iloc[:train_end]
            for i in indexes

                    bundle, masked_target, model_type=model_name,

            for metric in ("prediction_error", "directional_accuracy", "expectancy", "stability", "episode_count",
                           "episode_expectancy_std", "recurrence_consistency"):
                bundle_metrics[f"{prefix}_{metric}"] = result.get(metric, np.nan)
            score = calculate_fitness(result, complexity=sum(selected[i].expression.size for i in indexes),
                                      complexity_penalty=complexity_penalty)
            if np.isfinite(score):
                family = "linear" if model_name in {"ridge", "ols"} else model_name
                bundle_scores.setdefault(family, []).append(score)
        set_fitness_values = [float(np.mean(values)) for values in bundle_scores.values()]
        set_fitness = float(np.mean(set_fitness_values)) if set_fitness_values else -np.inf
        for i in indexes:
            selected[i] = replace(selected[i], fitness=set_fitness if np.isfinite(set_fitness) else selected[i].fitness,
                                  model_metrics={**selected[i].model_metrics, **bundle_metrics,
                                                 "feature_set_fitness": set_fitness})
            report_by_output[selected[i].output_column].update({"feature_set_fitness": set_fitness, **bundle_metrics})

    model = GeneticFeatureModel(
        selected=tuple(selected),
        input_columns=tuple(input_columns) if input_columns is not None else None,
        normalization_window=normalization_window,
        normalization_min_periods=normalization_min_periods,
        clip_z=clip_z,
    )

    report = pd.DataFrame(report_rows)
    if not report.empty:
        report = report.sort_values(["state_id", "target_column", "fitness"], ascending=[True, True, False]).reset_index(drop=True)
        report = select_validated_features(report, min_validation_ic=min_validation_ic)

                                                              "encoded_terminal_columns": list(variables),

                                                              "clip_z": clip_z})

"""Precomputed float32 temporal features for GP. Build once, pointer-lookup in loop."""

from __future__ import annotations

import numpy as np
import pandas as pd

from .jit_kernels import diff_kernel, lag_kernel, rolling_mean_kernel, rolling_std_kernel


class TemporalFeatureCache:
    """Contiguous float32 base + precomputed (var, window) temporal variants."""

    def __init__(self, index, base: dict[str, np.ndarray], windows: tuple[int, ...]):
        self.index = index
        self.n = len(index)
        self.base = base
        self.arrays: dict[str, np.ndarray] = dict(base)
        self.windows = tuple(windows)

    def get(self, key: str) -> np.ndarray:
        return self.arrays[key]

    def __contains__(self, key: str) -> bool:
        return key in self.arrays

    def prefix(self, n: int) -> "TemporalFeatureCache":
        """Zero-copy view over the first ``n`` rows.

        Every cached op is causal (lag/diff/rolling/window/block segment
        boundaries anchor at row 0), so the prefix of a full-frame build
        equals a build over the prefixed frame; slicing views adds no copy.
        """
        if not 0 <= n <= self.n:
            raise ValueError(f"prefix length must be in [0, {self.n}]; got {n}")
        if n == self.n:
            return self
        sub = TemporalFeatureCache(self.index[:n],
                                   {k: v[:n] for k, v in self.base.items()},
                                   self.windows)
        for k, v in self.arrays.items():
            if k not in sub.arrays:
                sub.arrays[k] = v[:n]
        return sub


def build_temporal_cache(inputs: pd.DataFrame, temporal_windows: tuple[int, ...] = (2, 4, 8, 16)) -> TemporalFeatureCache:
    """Precompute every (var, window) lag/rollmean/rollstd/diff/windowsum once."""
    wins = tuple(sorted(set(int(w) for w in temporal_windows)))
    if any(w <= 0 for w in wins):
        raise ValueError(f"temporal_windows must be positive; got {temporal_windows!r}")
    import polars as pl

    base: dict[str, np.ndarray] = {}
    try:
        plf = pl.from_pandas(pd.DataFrame(inputs))
    except Exception:
        plf = None
    if plf is not None:
        for c in plf.columns:
            try:
                col = plf[c].cast(pl.Float32, strict=False).to_numpy()
            except Exception:
                col = pd.to_numeric(inputs[c], errors="coerce").to_numpy(dtype=np.float32)
            base[str(c)] = np.ascontiguousarray(np.asarray(col, dtype=np.float32).reshape(-1))
    else:
        for c in list(inputs.columns):
            col = pd.to_numeric(inputs[c], errors="coerce").to_numpy(dtype=np.float32)
            base[str(c)] = np.ascontiguousarray(col.reshape(-1))
    cache = TemporalFeatureCache(inputs.index, base, wins)
    for name, arr in list(base.items()):
        xf = np.ascontiguousarray(arr.astype(np.float32, copy=False))
        for w in wins:
            cache.arrays["__lag__" + name + "__" + str(w)] = lag_kernel(xf, int(w))
            cache.arrays["__rollmean__" + name + "__" + str(w)] = rolling_mean_kernel(xf, int(w))
            cache.arrays["__rollstd__" + name + "__" + str(w)] = rolling_std_kernel(xf, int(w))
            cache.arrays["__diff__" + name + "__" + str(w)] = diff_kernel(xf, int(w))
            f = arr.astype(np.float64)
            n = len(f)
            ws = np.empty(n, dtype=np.float32)
            for i in range(n):
                if i + 1 < w:
                    ws[i] = np.nan
                    continue
                seg = f[i - w + 1: i + 1]
                ws[i] = np.float32(seg.sum()) if np.isfinite(seg).all() else np.nan
            cache.arrays["__windowsum__" + name + "__" + str(w)] = ws
    return cache



def evaluate_expression_cached(node, cache: TemporalFeatureCache, index: pd.Index) -> pd.Series:
    """Numpy evaluator with cache pointer-lookup; kernels for computed children."""
    from . import grammar as _g

    def ev(nd) -> np.ndarray:
        op = nd.op
        if op == "var":
            return cache.get(str(nd.value)).astype(np.float64, copy=False)
        if op == "const":
            return np.full(cache.n, float(nd.value), dtype=np.float64)
        if op == "partial":
            tmp = pd.DataFrame({k: cache.arrays[k] for k in cache.base.keys()}, index=index)
            return _g.evaluate(nd.children[0], tmp, str(nd.value))[1].to_numpy(dtype=np.float64)
        ch = [ev(c) for c in nd.children]
        x = ch[0]
        if op in ("lag", "diff", "derivative", "time_derivative"):
            lag = int(nd.value)
            if lag <= 0:
                raise ValueError(f"lag must be positive; got {lag!r}")
            b0 = nd.children[0]
            if b0.op == "var" and op in ("lag", "diff"):
                key = ("__lag__" if op == "lag" else "__diff__") + str(b0.value) + "__" + str(lag)
                if key in cache:
                    return cache.get(key).astype(np.float64)
            n = len(x)
            out = np.empty(n, dtype=np.float64)
            out[:lag] = np.nan
            if op == "lag":
                out[lag:] = x[:-lag]
            else:
                out[lag:] = x[lag:] - x[:-lag]
                if op in ("derivative", "time_derivative"):
                    out[lag:] = out[lag:] / float(lag)
            return _g._bounded(pd.Series(out, index=index)).to_numpy(dtype=np.float64)
        if op in ("rolling_mean", "window", "rolling_std", "block"):
            w = int(nd.value)
            if w <= 0:
                raise ValueError(f"window must be positive; got {w!r}")
            b0 = nd.children[0]
            if b0.op == "var":
                km = {"rolling_mean": "__rollmean__", "window": "__windowsum__", "rolling_std": "__rollstd__"}
                if op in km:
                    key = km[op] + str(b0.value) + "__" + str(w)
                    if key in cache:
                        return cache.get(key).astype(np.float64)
            xf = np.ascontiguousarray(x.astype(np.float32))
            if op == "rolling_mean":
                return rolling_mean_kernel(xf, w).astype(np.float64)
            if op == "window":
                f = x.astype(np.float64)
                n = len(f)
                ws = np.empty(n, dtype=np.float64)
                for i in range(n):
                    if i + 1 < w:
                        ws[i] = np.nan
                        continue
                    seg = f[i - w + 1: i + 1]
                    ws[i] = seg.sum() if np.isfinite(seg).all() else np.nan
                return ws
            if op == "rolling_std":
                return rolling_std_kernel(xf, w).astype(np.float64)
            n = len(x)
            out = np.empty(n, dtype=np.float64)
            for s in range(0, n, w):
                e = min(s + w, n)
                seg = x[s:e]
                cs = np.cumsum(np.where(np.isfinite(seg), seg, 0.0))
                cn = np.cumsum(np.isfinite(seg).astype(np.int64))
                out[s:e] = np.where(cn > 0, cs / np.maximum(cn, 1), np.nan)
            return out
        if op == "neg":
            return -x
        if op == "abs":
            return np.abs(x)
        if op == "signed_log":
            return np.sign(x) * np.log1p(np.abs(x))
        if op == "log":
            return _g._bounded(pd.Series(np.log(np.clip(np.abs(x), 1e-12, None)), index=index)).to_numpy(dtype=np.float64)
        if op == "sqrt":
            return _g._bounded(pd.Series(np.sqrt(np.abs(x)), index=index)).to_numpy(dtype=np.float64)
        if op == "signed_sqrt":
            return _g._bounded(pd.Series(np.sign(x) * np.sqrt(np.abs(x)), index=index)).to_numpy(dtype=np.float64)
        y = ch[1]
        if op == "add":
            return _g._bounded(pd.Series(x + y, index=index)).to_numpy(dtype=np.float64)
        if op == "sub":
            return _g._bounded(pd.Series(x - y, index=index)).to_numpy(dtype=np.float64)
        if op == "mul":
            return _g._bounded(pd.Series(x * y, index=index)).to_numpy(dtype=np.float64)
        if op == "div":
            den = np.where(np.abs(y) >= 1e-12, y, np.nan)
            return _g._bounded(pd.Series(x / den, index=index)).to_numpy(dtype=np.float64)
        if op == "power":
            e = np.clip(np.round(y), -4, 4)
            with np.errstate(all="ignore"):
                return _g._bounded(pd.Series(np.power(x, e), index=index)).to_numpy(dtype=np.float64)
        raise ValueError("unknown GP operator " + repr(op))

    vals = ev(node)
    vals = np.where(np.isfinite(vals), vals, np.nan)
    vals = np.clip(vals, -1e6, 1e6)
    return pd.Series(vals, index=index)

from __future__ import annotations

import json

from typing import Any

SCHEMA_VERSION = 1


def build_feature_library(selected_features=(), *, metadata=None):
    library = {"schema_version": SCHEMA_VERSION, "metadata": dict(metadata or {}), "regimes": {}}
    for item in selected_features:
        state = str(item.state_id)
        library["regimes"].setdefault(state, []).append({
            "target_column": item.target_column,
            "output_column": item.output_column,
            "expression": item.expression,
            "train_ic": float(item.train_ic),
            "validation_ic": float(item.validation_ic),
            "fitness": float(item.fitness),
            "direction": int(item.direction),
            "n_train": int(item.n_train),
            "n_validation": int(item.n_validation),
            "model_metrics": dict(getattr(item, "model_metrics", {}) or {}),
        })
    return library


def select_validated_features(report, *, min_fitness=None, min_validation_ic=None):
    if report is None or report.empty:
        return report.copy() if report is not None else report
    selected = report.copy()
    if min_fitness is not None and "fitness" in selected:
        selected = selected[selected["fitness"] >= min_fitness]
    if min_validation_ic is not None and "validation_ic" in selected:
        selected = selected[selected["validation_ic"].abs() >= min_validation_ic]
    return selected.reset_index(drop=True)


def _json_default(value):
    if hasattr(value, "__dataclass_fields__"):
        return {key: getattr(value, key) for key in value.__dataclass_fields__}
    raise TypeError(f"cannot serialize {type(value).__name__}")


def save_feature_library(library, path):
    target = Path(path)
    target.parent.mkdir(parents=True, exist_ok=True)
    data = dict(library)
    data.setdefault("schema_version", SCHEMA_VERSION)
    target.write_text(json.dumps(data, default=_json_default, indent=2), encoding="utf-8")
    return target


def load_feature_library(path):
    library = json.loads(Path(path).read_text(encoding="utf-8"))
    if library.get("schema_version") != SCHEMA_VERSION:
        raise ValueError(f"unsupported feature-library schema: {library.get('schema_version')}")
    for records in library.get("regimes", {}).values():
        for record in records:
            if isinstance(record.get("expression"), dict):
                record["expression"] = node_from_dict(record["expression"])
    return library


def get_features_for_regime(library, regime_id):
    return list(library.get("regimes", {}).get(str(regime_id), ()))


def register_regime_features(library, regime_id, features, *, replace=True):
    result = {**library, "regimes": dict(library.get("regimes", {}))}
    key = str(regime_id)
    existing = [] if replace else list(result["regimes"].get(key, ()))
    result["regimes"][key] = existing + list(features)
    return result

"""Safe expression grammar and causal time-series operators for GP."""

from __future__ import annotations

from dataclasses import dataclass
import numpy as np
import pandas as pd


_ARITHMETIC = ("add", "sub", "mul", "div", "power")
_UNARY = ("neg", "abs", "log", "sqrt", "signed_log", "signed_sqrt")
_TEMPORAL = ("lag", "diff", "derivative", "time_derivative", "rolling_mean", "rolling_std", "window", "block")
_CALCULUS = ("partial",)


def get_primitive_features(frame: pd.DataFrame | None = None, columns=None) -> tuple[str, ...]:
    """Return numeric causal terminals, excluding labels, IDs, and future data."""
    forbidden = ("future_", "target", "label", "barrier", "latent", "trade_allowed", "embedding")
    if columns is None:
        if frame is None:
            return ()
        columns = [c for c in frame if pd.api.types.is_numeric_dtype(frame[c])]
    result = tuple(str(c) for c in columns if not any(x in str(c).lower() for x in forbidden)
                   and "timestamp" not in str(c).lower() and not str(c).lower().endswith("_id"))
    return result


def get_operators() -> tuple[str, ...]:
    """List operators supported by the safe expression interpreter."""
    return _ARITHMETIC + _UNARY + _TEMPORAL + _CALCULUS


def get_operator_rules() -> dict[str, dict[str, object]]:
    """Describe arity and recursion policy; temporal/calculus operators are single-use."""
    rules = {op: {"arity": 2, "recursive": True, "family": "arithmetic"} for op in _ARITHMETIC}
    rules.update({op: {"arity": 1, "recursive": True, "family": "arithmetic"} for op in _UNARY})
    rules.update({op: {"arity": 1, "recursive": False, "family": "temporal"} for op in _TEMPORAL})
    rules.update({op: {"arity": 1, "recursive": False, "family": "calculus"} for op in _CALCULUS})
    return rules


def calculate_expression_depth(node: "Node") -> int:
    return node.height


def calculate_expression_complexity(node: "Node") -> int:
    return node.size


def validate_expression(node: "Node", primitive_features=None, max_depth: int = 10) -> bool:
    """Validate shape, terminals, depth, and restricted temporal recursion."""
    rules = get_operator_rules()
    allowed = set(primitive_features) if primitive_features is not None else None
    if node.height > max_depth:
        return False
    def visit(current: Node, restricted_parent: bool = False) -> bool:
        if current.op == "var":
            return bool(current.value is not None and (allowed is None or str(current.value) in allowed))
        if current.op == "const":
            try:
                return np.isfinite(float(current.value))
            except (TypeError, ValueError):
                return False
        rule = rules.get(current.op)
        if rule is None or len(current.children) != rule["arity"] or (restricted_parent and current.op in _TEMPORAL + _CALCULUS):
            return False
        if current.op in _TEMPORAL and (not isinstance(current.value, (int, np.integer)) or int(current.value) < 1):
            return False
        restricted_branch = restricted_parent or current.op in _TEMPORAL + _CALCULUS
        return all(visit(child, restricted_branch) for child in current.children)
    return visit(node)


@dataclass(frozen=True)
class Node:
    """One immutable node in an expression tree."""

    op: str
    children: tuple["Node", ...] = ()
    value: str | float | int | None = None

    @property
    def height(self) -> int:
        return 1 + max((child.height for child in self.children), default=0)

    @property
    def size(self) -> int:
        return 1 + sum(child.size for child in self.children)

    def __str__(self) -> str:
        if self.op == "var":
            return str(self.value)
        if self.op == "const":
            return f"{float(self.value):g}"
        if self.op in _TEMPORAL + _CALCULUS:
            if self.op == "partial":
                return f"partial[{self.value}]({self.children[0]})"
            return f"{self.op}[{self.value}]({self.children[0]})"
        if self.op in _UNARY:
            return f"{self.op}({self.children[0]})"
        symbols = {"add": "+", "sub": "-", "mul": "*", "div": "/", "power": "**"}
        if self.op in symbols:
            return f"({self.children[0]} {symbols[self.op]} {self.children[1]})"
        return f"{self.op}({', '.join(map(str, self.children))})"


def node_to_dict(node: Node) -> dict:
    return {"op": node.op, "value": node.value,
            "children": [node_to_dict(child) for child in node.children]}


def node_from_dict(data: dict) -> Node:
    return Node(op=str(data["op"]), value=data.get("value"),
                children=tuple(node_from_dict(child) for child in data.get("children", ())))


_MAX_MAGNITUDE = 1e6
_EPSILON = 1e-8


def _bounded(values: pd.Series) -> pd.Series:
    return values.replace([np.inf, -np.inf], np.nan).clip(-_MAX_MAGNITUDE, _MAX_MAGNITUDE)


def _safe_divide(numerator: pd.Series, denominator: pd.Series) -> pd.Series:
    safe = denominator.where(denominator.abs() >= _EPSILON)
    return numerator.div(safe)


def _rolling(node_op: str, values: pd.Series, window: int) -> pd.Series:
    # LEGACY fallback only; hot path uses TemporalFeatureCache + rolling_*_kernel (prange, float32).
    # Kept for audit/unit tests. Do not call inside evolution loop.
    rolling = values.rolling(window=window, min_periods=window)
    if node_op == "rolling_mean":
        return rolling.mean()
    if node_op == "rolling_std":
        return rolling.std(ddof=0)
    if node_op == "window":
        return rolling.sum()
    raise ValueError(f"unknown rolling operator {node_op!r}")


def evaluate(node: Node, inputs: pd.DataFrame, wrt: str | None = None) -> tuple[pd.Series, pd.Series]:
    """Evaluate a tree and its partial derivative with respect to one input.

    The derivative propagates analytically through pointwise and temporal
    operators. Temporal operators use the same causal window on values and
    tangents; rolling standard deviation uses its exact population-variance
    derivative. No source-code evaluation or future observation is used.
    """
    if wrt is None:
        return _evaluate_value(node, inputs), pd.Series(0.0, index=inputs.index, dtype=np.float64)
    if node.op == "var":
        name = str(node.value)
        if name not in inputs:
            raise KeyError(f"missing GP input column {name!r}")
        value = pd.to_numeric(inputs[name], errors="coerce").astype(np.float64)
        tangent = pd.Series(1.0 if name == wrt else 0.0, index=inputs.index, dtype=np.float64)
        return value, tangent
    if node.op == "const":
        value = pd.Series(float(node.value), index=inputs.index, dtype=np.float64)
        return value, pd.Series(0.0, index=inputs.index, dtype=np.float64)

    if node.op == "partial":
        _, partial = evaluate(node.children[0], inputs, str(node.value))
        # Second-order derivatives are not part of this grammar.
        return _bounded(partial), pd.Series(0.0, index=inputs.index, dtype=np.float64)

    values = [evaluate(child, inputs, wrt) for child in node.children]
    (x, dx) = values[0]
    if node.op in {"lag", "diff", "derivative"}:
        lag = int(node.value)
        if node.op == "lag":
            return x.shift(lag), dx.shift(lag)
        value = x - x.shift(lag)
        tangent = dx - dx.shift(lag)
        if node.op == "derivative":
            value = value / lag
            tangent = tangent / lag
        return _bounded(value), _bounded(tangent)
    if node.op == "time_derivative":
        lag = int(node.value)
        delta = inputs.index.to_series().diff().dt.total_seconds() if isinstance(inputs.index, pd.DatetimeIndex) else pd.Series(1.0, index=inputs.index)
        denominator = delta.replace(0.0, np.nan).rolling(lag, min_periods=lag).sum()
        return _bounded(x.diff(lag).div(denominator)), _bounded(dx.diff(lag).div(denominator))
    if node.op in {"rolling_mean", "window", "block"}:
        window = int(node.value)
        if node.op == "block":
            groups = np.arange(len(x)) // window
            return (_bounded(x.groupby(groups).transform(lambda s: s.expanding().mean())),
                    _bounded(dx.groupby(groups).transform(lambda s: s.expanding().mean())))
        rolling_op = "window" if node.op == "window" else "rolling_mean"
        return _rolling(rolling_op, x, window), _rolling(rolling_op, dx, window)
    if node.op == "rolling_std":
        window = int(node.value)
        rolling = x.rolling(window=window, min_periods=window)
        mean_x = rolling.mean()
        mean_dx = dx.rolling(window=window, min_periods=window).mean()
        mean_xdx = (x * dx).rolling(window=window, min_periods=window).mean()
        value = rolling.std(ddof=0)
        tangent = _safe_divide(mean_xdx - mean_x * mean_dx, value)
        return _bounded(value), _bounded(tangent)
    if node.op == "neg":
        return -x, -dx
    if node.op == "abs":
        return x.abs(), np.sign(x) * dx
    if node.op == "signed_log":
        return np.sign(x) * np.log1p(x.abs()), _safe_divide(dx, 1.0 + x.abs())
    if node.op == "log":
        return _bounded(np.log(x.abs().clip(lower=_EPSILON))), _bounded(_safe_divide(dx, x))
    if node.op == "sqrt":
        value = np.sqrt(x.abs())
        return _bounded(value), _bounded(np.sign(x) * _safe_divide(dx, 2.0 * value.clip(lower=_EPSILON)))
    if node.op == "signed_sqrt":
        value = np.sign(x) * np.sqrt(x.abs())
        tangent = _safe_divide(dx, 2.0 * np.sqrt(x.abs()).clip(lower=_EPSILON))
        return _bounded(value), _bounded(tangent)
    if node.op in {"add", "sub", "mul", "div", "power"}:
        y, dy = values[1]
        if node.op == "add":
            return _bounded(x + y), _bounded(dx + dy)
        if node.op == "sub":
            return _bounded(x - y), _bounded(dx - dy)
        if node.op == "mul":
            return _bounded(x * y), _bounded(dx * y + x * dy)
        if node.op == "power":
            exponent = y.round().clip(-4, 4)
            value = x.pow(exponent)
            # Exponents are rounded to a small integer in this protected
            # operator, so their derivative is intentionally zero.
            tangent = exponent * x.pow(exponent - 1) * dx
            return _bounded(value), _bounded(tangent)
        value = _safe_divide(x, y)
        tangent = _safe_divide(dx * y - x * dy, y.pow(2))
        return _bounded(value), _bounded(tangent)
    raise ValueError(f"unknown GP operator {node.op!r}")


def evaluate_expression(node: Node, inputs: pd.DataFrame) -> pd.Series:

    return evaluate(node, inputs)[0]


def _evaluate_value(node: Node, inputs: pd.DataFrame) -> pd.Series:
    """Fast value-only interpreter used by GP scoring (no tangent propagation)."""
    if node.op == "var":
        name = str(node.value)
        if name not in inputs:
            raise KeyError(f"missing GP input column {name!r}")
        return pd.to_numeric(inputs[name], errors="coerce").astype(np.float64)
    if node.op == "const":
        return pd.Series(float(node.value), index=inputs.index, dtype=np.float64)
    if node.op == "partial":
        return _bounded(evaluate(node.children[0], inputs, wrt=str(node.value))[1])
    values = [_evaluate_value(child, inputs) for child in node.children]
    x = values[0]
    if node.op == "lag":
        return x.shift(int(node.value))
    if node.op in {"diff", "derivative"}:
        lag = int(node.value)
        result = x.diff(lag)
        return _bounded(result / lag) if node.op == "derivative" else _bounded(result)
    if node.op == "time_derivative":
        lag = int(node.value)
        if isinstance(inputs.index, pd.DatetimeIndex):
            delta = inputs.index.to_series().diff().dt.total_seconds()
        else:
            delta = pd.Series(1.0, index=inputs.index)
        denom = delta.replace(0.0, np.nan).rolling(lag, min_periods=lag).sum()
        return _bounded(x.diff(lag).div(denom))
    if node.op in {"rolling_mean", "window", "rolling_std"}:
        return _rolling(node.op, x, int(node.value))
    if node.op == "block":
        window = int(node.value)
        groups = np.arange(len(x)) // window
        return _bounded(x.groupby(groups).transform(lambda s: s.expanding().mean()))
    if node.op == "neg":
        return -x
    if node.op == "abs":
        return x.abs()
    if node.op == "signed_log":
        return np.sign(x) * np.log1p(x.abs())
    if node.op == "log":
        return _bounded(np.log(x.abs().clip(lower=_EPSILON)))
    if node.op == "sqrt":
        return _bounded(np.sqrt(x.abs()))
    if node.op == "signed_sqrt":
        return _bounded(np.sign(x) * np.sqrt(x.abs()))
    y = values[1]
    if node.op == "add":
        return _bounded(x + y)
    if node.op == "sub":
        return _bounded(x - y)
    if node.op == "mul":
        return _bounded(x * y)
    if node.op == "div":
        return _bounded(_safe_divide(x, y))
    if node.op == "power":
        return _bounded(x.pow(y.round().clip(-4, 4)))
    raise ValueError(f"unknown GP operator {node.op!r}")

def random_tree(
    rng: np.random.Generator,
    variables: tuple[str, ...],
    max_depth: int,
    temporal_windows: tuple[int, ...] = (2, 4, 8, 16),
    grow: bool = True,
    allow_partial: bool = True,
    allow_temporal: bool = True,
) -> Node:
    """Create a bounded-depth tree using the configured grammar."""
    if not variables:
        raise ValueError("at least one GP input variable is required")
    if max_depth <= 1 or (grow and rng.random() < 0.32):
        return _random_terminal(rng, variables)

    binary_ops = ("add", "sub", "mul", "div", "power")
    unary_ops = ("neg", "abs", "log", "sqrt", "signed_log", "signed_sqrt") + (_TEMPORAL if allow_temporal else ())
    if rng.random() < 0.42:
        op = str(rng.choice(binary_ops))
        return Node(op, (
            random_tree(rng, variables, max_depth - 1, temporal_windows, grow, allow_partial, allow_temporal),
            random_tree(rng, variables, max_depth - 1, temporal_windows, grow, allow_partial, allow_temporal),
        ))

    if allow_partial and rng.random() < 0.12:
        child = random_tree(rng, variables, max_depth - 1, temporal_windows, grow, allow_partial=False, allow_temporal=False)
        eligible = variables_in(child)
        if eligible and _has_nonlinear_operation(child):
            return Node("partial", (child,), value=str(rng.choice(eligible)))
        return child

    op = str(rng.choice(unary_ops))
    # Restricted operators wrap an arithmetic-only subtree, preventing chains of
    # temporal transforms while retaining recursive arithmetic below them.
    restricted = op in _TEMPORAL + _CALCULUS
    child = random_tree(rng, variables, max_depth - 1, temporal_windows, grow, allow_partial=False, allow_temporal=False) if restricted else random_tree(rng, variables, max_depth - 1, temporal_windows, grow, allow_partial, allow_temporal)
    parameter = int(rng.choice(temporal_windows)) if op in _TEMPORAL else None
    return Node(op, (child,), value=parameter)


def node_at_path(node: Node, path: tuple[int, ...]) -> Node:

    children = list(node.children)

"""Numba-compiled hot kernels for the GP search.

Every kernel is ``nogil=True`` so the (state, target) thread pool in
``gp/parallel.py`` runs them truly in parallel (pandas ``rank`` holds the GIL).
Results match the previous pandas implementations to float precision.
If numba is not installed the same functions run as plain Python (slow, but
correct), so imports never break.
"""

from __future__ import annotations

import numpy as np

try:  # pragma: no cover - exercised implicitly
    from numba import njit, prange
except ImportError:  # graceful fallback
    def njit(*args, **kwargs):
        if len(args) == 1 and callable(args[0]):
            return args[0]
        return lambda fn: fn

    def prange(*args):
        return range(*args)

__all__ = ["spearman_kernel", "blockwise_spearman_kernel", "abs_corr_exceeds", "batch_spearman_kernel", "rolling_mean_kernel", "rolling_std_kernel", "lag_kernel", "diff_kernel", "quantize_tick_kernel", "barrier_sim_kernel", "excursion_kernel", "elapsed_err_kernel", "barrier_outcome_kernel"]


@njit(cache=True, nogil=True)
def _average_ranks(values):
    # NOTE: kept as serial core; parallel fan-out happens in spearman_kernel / batch kernel via prange.
    n = values.shape[0]
    order = np.argsort(values, kind="mergesort")
    ranks = np.empty(n, dtype=np.float64)
    i = 0
    while i < n:
        j = i
        while j + 1 < n and values[order[j + 1]] == values[order[i]]:
            j += 1
        avg = 0.5 * (i + j) + 1.0  # 1-based average rank of the tie group
        for k in range(i, j + 1):
            ranks[order[k]] = avg
        i = j + 1
    return ranks


@njit(cache=True, nogil=True)
def spearman_kernel(left, right):
    """Return (rho, count) over rows where both inputs are finite; rho=nan if undefined."""
    n = left.shape[0]
    valid = np.empty(n, dtype=np.bool_)
    count = 0
    for i in range(n):
        ok = np.isfinite(left[i]) and np.isfinite(right[i])
        valid[i] = ok
        if ok:
            count += 1
    if count < 2:
        return np.nan, count
    x = np.empty(count, dtype=np.float64)
    y = np.empty(count, dtype=np.float64)
    k = 0
    for i in range(n):
        if valid[i]:
            x[k] = left[i]
            y[k] = right[i]
            k += 1
    rx = _average_ranks(x)
    ry = _average_ranks(y)
    mx = rx.mean()
    my = ry.mean()
    sxx = 0.0
    syy = 0.0
    sxy = 0.0
    for i in range(count):
        dx = rx[i] - mx
        dy = ry[i] - my
        sxx += dx * dx
        syy += dy * dy
        sxy += dx * dy
    denom = np.sqrt(sxx * syy)
    if denom <= 1e-15:
        return np.nan, count
    return sxy / denom, count


@njit(cache=True, nogil=True)
def blockwise_spearman_kernel(values, target, train_mask, train_end, block_count, min_block_samples):
    """Spearman IC per chronological block of ``[0, train_end)`` (np.array_split rules)."""
    out = np.empty(block_count, dtype=np.float64)
    used = 0
    base = train_end // block_count
    extra = train_end % block_count
    start = 0
    for b in range(block_count):
        size = base + (1 if b < extra else 0)
        stop = start + size
        m = 0
        for i in range(start, stop):
            if train_mask[i]:
                m += 1
        if m >= min_block_samples:
            xs = np.empty(m, dtype=np.float64)
            ys = np.empty(m, dtype=np.float64)
            k = 0
            for i in range(start, stop):
                if train_mask[i]:
                    xs[k] = values[i]
                    ys[k] = target[i]
                    k += 1
            rho, cnt = spearman_kernel(xs, ys)
            if cnt >= min_block_samples and np.isfinite(rho):
                out[used] = rho
                used += 1
        start = stop
    return out[:used]


@njit(cache=True, nogil=True)
def abs_corr_exceeds(candidate, prior, mask, threshold, min_samples):
    """True if |Pearson(candidate, prior)| >= threshold on finite masked rows."""
    n = candidate.shape[0]
    count = 0
    for i in range(n):
        if mask[i] and np.isfinite(candidate[i]) and np.isfinite(prior[i]):
            count += 1
    if count < min_samples:
        return False
    sa = 0.0
    sb = 0.0
    for i in range(n):
        if mask[i] and np.isfinite(candidate[i]) and np.isfinite(prior[i]):
            sa += candidate[i]
            sb += prior[i]
    ma = sa / count
    mb = sb / count
    saa = 0.0
    sbb = 0.0
    sab = 0.0
    for i in range(n):
        if mask[i] and np.isfinite(candidate[i]) and np.isfinite(prior[i]):
            da = candidate[i] - ma
            db = prior[i] - mb
            saa += da * da
            sbb += db * db
            sab += da * db
    if saa / count <= 1e-24 or sbb / count <= 1e-24:  # std <= 1e-12
        return False
    corr = sab / np.sqrt(saa * sbb)
    return bool(np.isfinite(corr) and abs(corr) >= threshold)


@njit(cache=True, nogil=True, parallel=True)
def batch_spearman_kernel(vals, tgts, min_samples):
    """Batched (rho, n) over stacked candidate matrix. Heavy: parallel fan-out."""
    n_cand = vals.shape[0]
    out_rho = np.empty(n_cand, dtype=np.float64)
    out_n = np.empty(n_cand, dtype=np.int64)
    for c in prange(n_cand):
        rho, cnt = spearman_kernel(vals[c], tgts[c])
        if cnt < min_samples or not np.isfinite(rho):
            out_rho[c] = np.nan
        else:
            out_rho[c] = rho
        out_n[c] = cnt
    return out_rho, out_n


@njit(cache=True, nogil=True, parallel=True)
def rolling_mean_kernel(x, window):
    """Causal rolling mean, min_periods=window. Heavy: parallel."""
    n = x.shape[0]
    out = np.empty(n, dtype=np.float32)
    for i in prange(n):
        if i + 1 < window:
            out[i] = np.nan
            continue
        s = 0.0
        c = 0
        for j in range(i - window + 1, i + 1):
            v = x[j]
            if np.isfinite(v):
                s += v
                c += 1
        if c < window:
            out[i] = np.nan
        else:
            out[i] = np.float32(s / c)
    return out


@njit(cache=True, nogil=True, parallel=True)
def rolling_std_kernel(x, window):
    """Causal population rolling std (ddof=0), min_periods=window. Heavy: parallel."""
    n = x.shape[0]
    out = np.empty(n, dtype=np.float32)
    for i in prange(n):
        if i + 1 < window:
            out[i] = np.nan
            continue
        s = 0.0
        s2 = 0.0
        c = 0
        for j in range(i - window + 1, i + 1):
            v = x[j]
            if np.isfinite(v):
                s += v
                s2 += v * v
                c += 1
        if c < window:
            out[i] = np.nan
        else:
            m = s / c
            var = s2 / c - m * m
            out[i] = np.float32(np.sqrt(var)) if var > 0.0 else np.float32(0.0)
    return out


@njit(cache=True, nogil=True)
def lag_kernel(x, lag):

    n = x.shape[0]
    out = np.empty(n, dtype=np.float32)
    for i in range(n):

@njit(cache=True, nogil=True)
def diff_kernel(x, lag):
    """Causal diff x[i]-x[i-lag]; NaN for i < lag. Lightweight: serial."""

def _tick_decimals(tick) -> int:
    """Decimal places implied by Decimal(str(tick)) -- the exponent .quantize() uses.

    quantize() ignores the tick's magnitude entirely: str(0.01) -> 2 decimals,
    str(1.0) -> 1 decimal (grid 0.1), str(100) -> 0. This mirrors that rule
    exactly for int and float ticks alike.
    """
    from decimal import Decimal
    return max(0, -int(Decimal(str(tick)).as_tuple().exponent))


def quantize_tick_kernel(price, tick):
    """Tick snap with exact Decimal(str(v)).quantize(str(tick), ROUND_HALF_UP) parity.

    Decimal .quantize() rounds to the tick's decimal EXPONENT (grid = 10^-d,
    d from str(tick)), ties (exact .5) away from zero. The njit kernel
    computes k = floor(|v|*10^d + 0.5 + bias) with bias = 1e-6 + count*1.1e-15
    absorbing float error of v*10^d (canonical .5-tie doubles land within
    ~half-ulp of the tie; values in that sub-tie window also resolve up --
    indistinguishable from the tie in float64), then k/10^d is correctly
    rounded to the nearest grid double = float(Decimal) bit for bit.
    Non-finite values pass through. The wrapper resolves d from str(tick)
    before entering numba (str() is unavailable in nopython mode).
    """
    arr = np.ascontiguousarray(np.asarray(price, dtype=np.float64).reshape(-1))
    return _quantize_tick_kernel(arr, _tick_decimals(tick))


@njit(cache=True, nogil=True)
def _quantize_tick_kernel(price, decimals):
    """Numba core: round-half-up to the 10^-decimals grid; ties away from zero."""
    n = price.shape[0]
    out = np.empty(n, dtype=np.float64)
    s = 10.0 ** decimals
    for i in range(n):
        v = price[i]
        if not np.isfinite(v):
            out[i] = v
            continue
        c = np.abs(v) * s
        k = np.floor(c + 0.5 + c * 1.1e-15 + 1e-6)
        out[i] = (k / s) if v >= 0.0 else -(k / s)
    return out


def quantize_tick_scalar(v, tick: float) -> float:
    """Python twin of quantize_tick_kernel: same str(tick) decimals, same IEEE
    op order, same Decimal ROUND_HALF_UP parity (ties away from zero).

    Non-numeric inputs and non-finite floats pass through unchanged (the
    Decimal path also leaves non-finite values alone).
    """
    import math
    try:
        f = float(v)
    except (TypeError, ValueError):
        return v
    if not math.isfinite(f):
        return v
    s = 10.0 ** _tick_decimals(tick)
    c = abs(f) * s
    k = math.floor(c + 0.5 + c * 1.1e-15 + 1e-6)
    return (k / s) if f >= 0.0 else -(k / s)


@njit(cache=True, nogil=True, parallel=True)
def barrier_sim_kernel(entry, high, low, tp_w, sl_w, timeout, side):
    """Per-event TP/SL/timeout sim. Heavy O(E*H): parallel over events.

    Matches simulate_barriers semantics: side-aware barriers, stop wins ties,
    timeout after timeout bars, exit = last available entry when timed out.
    """
    n = entry.shape[0]
    outcome = np.empty(n, dtype=np.int64)
    exit_px = np.empty(n, dtype=np.float64)
    bars = np.empty(n, dtype=np.int64)
    for e in prange(n):
        if not (np.isfinite(entry[e]) and np.isfinite(tp_w[e]) and np.isfinite(sl_w[e])):
            outcome[e] = 0
            exit_px[e] = np.nan
            bars[e] = 0
            continue
        s = side[e]
        p = entry[e]
        tp = p * (1.0 + tp_w[e]) if s > 0 else p * (1.0 - tp_w[e])
        sl = p * (1.0 - sl_w[e]) if s > 0 else p * (1.0 + sl_w[e])
        hit = 0
        xp = p
        lim = e + timeout
        if lim > n:
            lim = n
        for t in range(e, lim):
            h = high[t]
            lo = low[t]
            if s > 0:
                sl_hit = np.isfinite(lo) and lo <= sl
                tp_hit = np.isfinite(h) and h >= tp
            else:
                sl_hit = np.isfinite(h) and h >= sl
                tp_hit = np.isfinite(lo) and lo <= tp
            if sl_hit:
                hit = -1
                xp = sl
                break
            if tp_hit:
                hit = 1
                xp = tp
                break
        if hit == 0:
            xp = entry[lim - 1] if (lim - 1 >= e and np.isfinite(entry[lim - 1])) else p
        outcome[e] = hit
        exit_px[e] = xp
        bars[e] = lim - e
    return outcome, exit_px, bars



@njit(cache=True, nogil=True, parallel=True)
def excursion_kernel(entry, high, low, timeout, side):
    """Per-event max favorable/adverse excursion fractions. Heavy: parallel.

    Matches forward_excursions semantics: future path is bars e+1..e+H
    (entry bar excluded); tail uses whatever future bars remain.
    """
    n = entry.shape[0]
    mfe = np.empty(n, dtype=np.float64)
    mae = np.empty(n, dtype=np.float64)
    for e in prange(n):
        p = entry[e]
        if not np.isfinite(p) or p == 0.0:
            mfe[e] = np.nan
            mae[e] = np.nan
            continue
        best_up = 0.0
        best_dn = 0.0
        lim = e + timeout + 1
        if lim > n:
            lim = n
        for t in range(e + 1, lim):
            h = high[t]
            lo = low[t]
            if np.isfinite(h):
                r = (h - p) / p
                if r > best_up:
                    best_up = r
            if np.isfinite(lo):
                r = (lo - p) / p
                if r < best_dn:
                    best_dn = r
        if side[e] > 0:
            mfe[e] = best_up
            mae[e] = -best_dn
        else:
            mfe[e] = -best_dn
            mae[e] = best_up
    return mfe, mae


@njit(cache=True, nogil=True, parallel=True)
def elapsed_err_kernel(clock_ns, end_ns, resid, pos_idx):
    """Backward-asof gather: latest resid with end_ns <= clock. Heavy: parallel."""
    n = pos_idx.shape[0]
    m = end_ns.shape[0]
    out = np.empty(n, dtype=np.float64)
    for k in prange(n):
        t = clock_ns[pos_idx[k]]
        lo = 0
        hi = m - 1
        ans = -1
        while lo <= hi:
            mid = (lo + hi) // 2
            if end_ns[mid] <= t:
                ans = mid
                lo = mid + 1
            else:
                hi = mid - 1
        out[k] = 0.0 if ans < 0 else (resid[ans] if np.isfinite(resid[ans]) else 0.0)
    return out


@njit(cache=True, nogil=True, parallel=True)
def barrier_outcome_kernel(high, low, side, tp_level, sl_level, timeout):
    """TP/SL/timeout outcome for EVERY row. Exact mirror of the vectorized
    O(n*H) k-loop in outcome/triple_barrier.simulate_barriers, early-exiting
    per row and parallel over rows.

    Contract (must stay bit-identical to the k-loop):
      - path = bars e+1 .. min(e+timeout, n-1); the entry bar is excluded;
      - stop is checked before TP on each bar (stop wins ties);
      - raw IEEE comparisons (NaN level or NaN bar never hits, +inf hits);
      - both levels NaN (side-0 / unscored rows) -> outcome 0 immediately;
      - outcome: +1 TP first, -1 SL first (or tie), 0 timeout/no event.
    """
    n = side.shape[0]
    outcome = np.empty(n, dtype=np.int64)
    for e in prange(n):
        sl = sl_level[e]
        tp = tp_level[e]
        if np.isnan(sl) and np.isnan(tp):
            # comparisons against NaN are always False in the k-loop too
            outcome[e] = 0
            continue
        s = side[e]
        hit = 0
        for k in range(1, timeout + 1):
            t = e + k
            if t >= n:
                break
            h = high[t]
            lo = low[t]
            if s > 0.0:
                sl_hit = lo <= sl
                tp_hit = h >= tp
            else:
                sl_hit = h >= sl
                tp_hit = lo <= tp
            if sl_hit:
                hit = -1
                break
            if tp_hit:
                hit = 1
                break
        outcome[e] = hit
    return outcome

"""Pair-loop parallelism for offline GP search (threads, never per live row).

Two entry points:

* ``search_pairs_sequential`` -- the ORIGINAL loop (kept verbatim and
  bit-identical) for audits and single-core machines.
* ``search_pairs_threaded`` -- the same pair computations run with a
  ``ThreadPoolExecutor`` over independent (state, target) pairs.

Threading is justified by the repo-local micro-benchmark
(``_bench_gp_micro.py``, 16 cores): walk-forward ridge/ols fits speed up
~1.8x and pandas expression evaluation ~2-2.9x with 2-8 workers, because the
hot loops live in numpy/pandas C code that releases the GIL. Gains plateau
past 4 workers, so ``workers`` defaults to min(4, cpu_count).

Determinism contract
--------------------
The original search shares ONE ``np.random.default_rng(seed)`` across all
pairs, so run order (and worker completion order) would leak into results.
Both entry points therefore derive a per-pair generator from
``np.random.SeedSequence(seed)`` keyed by ``(state_id, target_column)``.
Sequential and threaded runs are bit-identical to each other; they differ
from the legacy shared-RNG stream (documented in the GP README caveat
"Run order affects results").
"""

from __future__ import annotations

from concurrent.futures import ThreadPoolExecutor
from typing import Any

import numpy as np


def derive_pair_seed(seed: int, state_id: str, target_column: str) -> int:
    """Deterministic per-pair seed; independent of pair evaluation order.

    ``np.random.SeedSequence`` entropy must be integers, so the string identity
    of the pair is folded in with a stable (non-salted) hash -- ``hash()`` is
    randomized per process and would break reproducibility across runs.
    """
    import hashlib

    digest = hashlib.blake2b(
        f"{int(seed)}|{state_id}|{target_column}".encode("utf-8"), digest_size=8
    ).digest()
    pair_tag = int.from_bytes(digest, "big")
    seq = np.random.SeedSequence([int(seed), pair_tag])
    return int(seq.generate_state(1, dtype=np.uint64)[0] % (2**63 - 1))


def default_worker_count(n_pairs: int) -> int:
    """Cap parallelism where the benchmark shows gains plateau (~4)."""
    import os

    cpu = os.cpu_count() or 4
    return int(max(1, min(4, cpu, max(1, n_pairs))))


def run_pairs(
    pair_fn,
    pairs: list[tuple[str, str]],
    *,
    workers: int = 1,
    worker_count_fn=default_worker_count,
    show_progress: bool = False,
) -> list[Any]:
    """Run ``pair_fn(state_id, target_column)`` over pairs, in input order.

    ``workers=1`` is the sequential path (legacy behavior, same order).
    ``workers>1`` uses a thread pool sized by ``worker_count_fn``; results
    are returned in input order regardless of completion order. ``pair_fn``
    must be thread-safe: no shared RNG, no shared mutable caches.
    """
    pairs = list(pairs)
    if not pairs:
        return []
    if workers is None or workers <= 1:
        return [pair_fn(state_id, target) for state_id, target in pairs]
    max_workers = worker_count_fn(len(pairs))
    if max_workers <= 1:
        return [pair_fn(state_id, target) for state_id, target in pairs]
    if show_progress:
        from tqdm.auto import tqdm

        pairs = tqdm(pairs, desc="GP pair search")
    with ThreadPoolExecutor(max_workers=max_workers) as pool:
        return list(pool.map(lambda pair: pair_fn(*pair), pairs))

def deduplicate_population(population):

def filter_constant_features(population, frame, min_unique=2, sample_mask=None, cache=None):
    """Drop expressions with < min_unique finite values (optionally on sample rows).

    Uses the caller's precomputed TemporalFeatureCache when given (the evolution
    hot loop passes its run-level cache); if the cache is None a default one is
    built once. A cache-evaluation failure falls back to the pandas evaluator so
    a cache gap can never silently drop a valid node.
    """
    mask = None
    if sample_mask is not None:
        mask = np.asarray(sample_mask, dtype=bool)
        if len(mask) != len(frame):
            raise ValueError("sample_mask must match the frame rows")
    if cache is None:
        try:
            from .feature_cache import build_temporal_cache as _bc
            cache = _bc(frame)
        except Exception:
            cache = None
    kept = []
    for node in population:
        try:
            if cache is not None:
                try:
                    values = evaluate_expression_cached(node, cache, frame.index).to_numpy(dtype=np.float64)
                except (ValueError, KeyError, TypeError, FloatingPointError, AttributeError, OverflowError):
                    values = evaluate_expression(node, frame).to_numpy(dtype=np.float64)
            else:
                values = evaluate_expression(node, frame).to_numpy(dtype=np.float64)
            if mask is not None:
                values = values[mask]
            values = values[np.isfinite(values)]
            if np.unique(values).size >= min_unique:
                kept.append(node)
        except (ValueError, KeyError, TypeError, FloatingPointError, AttributeError):
            continue
    return kept


def filter_redundant_features(population, frame, threshold=0.98, min_samples=30, sample_mask=None, cache=None):
    """Keep the first representative from each highly correlated expression group.

    Uses the caller's TemporalFeatureCache when given (pandas fallback on any
    cache failure). The mask is validated once up front and prior values are
    stored already-masked, so every stored series has identical length.
    """
    if not 0 < threshold <= 1:
        raise ValueError("threshold must be in (0, 1]")
    mask = np.ones(len(frame), dtype=bool) if sample_mask is None else np.asarray(sample_mask, dtype=bool)
    if len(mask) != len(frame):
        raise ValueError("sample_mask must match the frame rows")
    if cache is None:
        try:
            from .feature_cache import build_temporal_cache as _bc
            cache = _bc(frame)
        except Exception:
            cache = None
    from .jit_kernels import abs_corr_exceeds as _ex

    kept, values = [], []
    for node in tqdm(population, desc="Filtering Redundant Expressions", leave=False):
        try:
            if cache is not None:
                try:
                    series = evaluate_expression_cached(node, cache, frame.index).to_numpy(dtype=np.float64)
                except (ValueError, KeyError, TypeError, FloatingPointError, AttributeError, OverflowError):
                    series = evaluate_expression(node, frame).to_numpy(dtype=np.float64)
            else:
                series = evaluate_expression(node, frame).to_numpy(dtype=np.float64)
        except (ValueError, KeyError, TypeError, FloatingPointError, AttributeError):
            continue
        if len(series) != len(mask):
            raise ValueError("sample_mask must match the frame rows")
        series = series[mask]
        all_true = np.ones(series.size, dtype=bool)
        duplicate = any(
            bool(_ex(np.ascontiguousarray(series), np.ascontiguousarray(prior), all_true,
                     float(threshold), int(min_samples)))
            for prior in values
        )
        if not duplicate:
            kept.append(node)
            values.append(series)
    return kept


def initialize_population(
    rng: np.random.Generator,
    variables: tuple[str, ...],
    population_size: int,
    max_depth: int,
    temporal_windows: tuple[int, ...],
) -> list[Node]:
    """Create a mixed grow/full population with varied initial complexity."""
    if population_size < 2:
        raise ValueError("population_size must be >= 2")
    trees = []
    for index in range(population_size):
        depth = int(rng.integers(2, max_depth + 1)) if max_depth > 2 else max_depth
        trees.append(random_tree(
            rng,
            variables,
            depth,
            temporal_windows=temporal_windows,
            grow=(index % 2 == 0),
        ))

def tournament_select(
    rng: np.random.Generator,
    scored_population: list[tuple[Node, float]],
    tournament_size: int,
) -> Node:
    """Select the highest-fitness individual in a random tournament."""
    if not scored_population:
        raise ValueError("cannot select from an empty population")
    size = min(max(1, tournament_size), len(scored_population))
    indices = rng.choice(len(scored_population), size=size, replace=False)
    winner = max((scored_population[int(index)] for index in indices), key=lambda item: item[1])
    return winner[0]


def crossover(
    rng: np.random.Generator,
    left: Node,
    right: Node,
    max_depth: int,
) -> Node:
    """Replace one random subtree, retrying until the height bound holds."""
    left_paths = all_paths(left)
    right_paths = all_paths(right)
    for _ in range(12):
        left_path = left_paths[int(rng.integers(len(left_paths)))]
        right_path = right_paths[int(rng.integers(len(right_paths)))]
        child = replace_at_path(left, left_path, _node_at(right, right_path))
        if child.height <= max_depth and not has_nested_partial(child) and validate_expression(child, max_depth=max_depth):
            return child
    return left


def mutate(
    rng: np.random.Generator,
    node: Node,
    variables: tuple[str, ...],
    max_depth: int,
    temporal_windows: tuple[int, ...],
) -> Node:
    """Replace one subtree with a random bounded subtree."""
    paths = list(all_paths(node))
    rng.shuffle(paths)
    for path in paths:
        remaining_depth = max_depth - len(path)
        if remaining_depth < 1:
            continue
        current = node
        inside_partial = False
        for index in path:
            if current.op == "partial":
                inside_partial = True
            current = current.children[index]
        replacement = random_tree(
            rng,
            variables,
            remaining_depth,
            temporal_windows=temporal_windows,
            grow=True,
            allow_partial=not inside_partial,
        )
        child = replace_at_path(node, path, replacement)
        if child.height <= max_depth and not has_nested_partial(child) and validate_expression(child, variables, max_depth=max_depth):
            return child
    return node


def _node_at(node: Node, path: tuple[int, ...]) -> Node:
    current = node
    for index in path:
        current = current.children[index]
    return current
