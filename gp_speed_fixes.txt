def batch_spearman_kernel(vals, tgts, min_samples):

@njit(cache=True, nogil=True, parallel=True)
def rolling_std_kernel(x, window):
    n = x.shape[0]
    out = np.empty(n, dtype=np.float32)
    for i in prange(n):
        if i + 1 < window:
            out[i] = np.nan
            continue
        s = 0.0
        s2 = 0.0
        c = 0
        for j in range(i - window + 1, i + 1):
            v = x[j]
            if np.isfinite(v):
                s += v
                s2 += v * v
                c += 1
        if c == 0:
            out[i] = np.nan
        else:
            m = s / c
            var = s2 / c - m * m
            if var < 0.0:
                var = 0.0
            out[i] = np.float32(np.sqrt(var))
    return out

CHECK: there may be significant mathematical differences
- Old way (scipy.stats.rankdata + Pearson): scipy.stats.rankdata has built-in methods for handling ties (by default, it assigns the average of the ranks that would have been assigned to all the tied values).
- New way (_numba_spearman): Writing a tie-averaging rank algorithm inside Numba (nogil) is notoriously tricky because standard sorting algorithms change indices, and handling identical values requires special care to avoid bias or crashes. If _numba_spearman uses a simpler ranking method (like min, max, or dense ranking) or breaks ties arbitrarily via standard sorting, your correlation values (rho) for datasets with ties will change compared to SciPy's exact average-rank method.
- But performace here is rlly good

 def build_gp_inputs(features, input_columns=None, normalization_window=500, normalization_min_periods=50, clip_z=8.0):
    # polars normalization (multithreaded rolling_mean) -> float32 frame; kernels consume this once.
    import polars as _pl
    cols = list(input_columns) if input_columns is not None else [c for c in list(features.columns) if not _is_price_level_name(c)]
    if not cols:
        raise ValueError("no usable GP input columns (all columns look like price levels)")
    wide = {c: pd.to_numeric(features[c], errors="coerce").to_numpy(dtype=np.float32) for c in cols}
    try:
        plf = _pl.DataFrame({c: wide[c] for c in cols})
        out = {}
        for c in cols:
            s = plf[c].cast(_pl.Float32, strict=False)
            rm = s.rolling_mean(window_size=int(normalization_window), min_periods=int(normalization_min_periods))
            rs = s.rolling_std(window_size=int(normalization_window), min_periods=int(normalization_min_periods), ddof=0)
            z = ((s - rm) / rs.clip_min(1e-12)).clip(-float(clip_z), float(clip_z)).fill_null(0.0).fill_nan(0.0)
            out[c] = z.to_numpy().astype(np.float32)
        return pd.DataFrame(out, index=features.index)
    except Exception:
        frame = pd.DataFrame(wide, index=features.index)
        mean = frame.rolling(window=normalization_window, min_periods=normalization_min_periods).mean()
        std = frame.rolling(window=normalization_window, min_periods=normalization_min_periods).std(ddof=0)
        z = ((frame - mean) / std.clip(lower=1e-12)).clip(-clip_z, clip_z)
        return z.fillna(0.0).astype(np.float32)
