 702|     /// Finalize the completed 1-second (micro) bar: assemble the 15-feature
 703|     /// vector ΓÇö micro `price_roc_3` (fresh every second) + macro/regime and
 704|     /// orderflow features forward-filled from the last TIMEFRAME_BARRIER
 705|     /// rollover (mirrors `LiveFeatureEngine.compute_1s_feature_vector`).
 706|     fn finish_micro(&mut self, close: f64, close_ms: i64) -> Feature {
 707|         self.last_bar_close_ms = close_ms;
 708| 
 709|         // Micro price rate-of-change is deliberately NOT computed here on 1 s
 710|         // closes: the research state space computes `price_roc_3` on the 300 s
 711|         // COMPOSITE close series (`feature_builder.py: close.pct_change(3)`)
 712|         // and broadcasts it to every 1 s row. `m.price_roc_3` carries that
 713|         // macro ROC (computed in `compute_macro_features`). The model was
 714|         // trained on the macro ROC ΓÇö feeding a 1 s ROC is a research/live
 715|         // divergence that shifts the single most important model input.
 716|         self.micro_closes.push_back(close);
 717|         if self.micro_closes.len() > MAX_MICRO_CLOSES {
 718|             self.micro_closes.pop_front();
 719|         }
 720| 
 721|         // Capture this (just-closed) 1-second bar's surviving tick sign into the
 722|         // composite-bar autocorrelation series, then reset (features.py
 723|         // compute_1s_feature_vector appends `_second_last_tick_sign` and resets).
 724|         if self.orderflow_enabled() {
 725|             self.bar_tick_signs.push(self.second_last_tick_sign);
 726|             self.second_last_tick_sign = 0;
 727|         }
 728| 
 729|         let m = self.macro_features.unwrap_or_default();
 730| 
 731|         // --- Real model P(WIN) over the full 15-feature vector, in the exact
 732|         // order produced by ml_state_optimized.py / recorded in
 733|         // lgb_model.features.json: 7 base + 8 orderflow. Falls back to the
 734|         // deterministic logit proxy only when no model is loaded (so the
 735|         // engine is still usable offline / without an export). ---
 736|         let features: [f64; 15] = [
 737|             m.spread_bps_z,
 738|             m.ma_slope_5,
 739|             m.rsi_x_ma,
 740|             m.ma_accel,
 741|             m.price_roc_3,
 742|             m.rsi_2,
 743|             m.sma_14,
 744|             m.ofi_z,
 745|             m.cvd_delta_z,
 746|             m.cvd_slope_z,
 747|             m.absorption_z,
 748|             m.tick_autocorr,
 749|             m.inst_ratio,
 750|             m.delta_divergence_z,
 751|             m.depth_imbalance_z,
 752|         ];
 753|         let win_prob = match self.model.as_ref() {
 754|             Some(mdl) => mdl.predict_proba(&features),
 755|             None => self.proxy_win_prob(&m, m.price_roc_3, close),

 763|             spread_bps_z: m.spread_bps_z,

 765|             rsi_x_ma: m.rsi_x_ma,
